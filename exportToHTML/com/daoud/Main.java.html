1    package com.daoud;
2    
3    import com.daoud.db.DatabaseManager_online;
4    import com.daoud.ui.LoginScreen;
5    import javafx.application.Application;
6    import javafx.collections.ListChangeListener;
7    import javafx.geometry.NodeOrientation;
8    import javafx.stage.Stage;
9    import javafx.stage.Window;
10   import javax.swing.JOptionPane;
11   
12   public class Main extends Application {
13   
14       @Override
15       public void start(Stage stage) {
16           try {
17               Window.getWindows().addListener((ListChangeListener<Window>) change -> {
18                   while (change.next()) {
19                       if (change.wasAdded()) {
20                           for (Window window : change.getAddedSubList()) {
21                               if (window.getScene() != null && window.getScene().getRoot() != null) {
22                                   window.getScene().getRoot()
23                                           .setNodeOrientation(NodeOrientation.RIGHT_TO_LEFT);
24                               }
25                               window.sceneProperty().addListener((obs, oldScene, newScene) -> {
26                                   if (newScene != null && newScene.getRoot() != null) {
27                                       newScene.getRoot()
28                                               .setNodeOrientation(NodeOrientation.RIGHT_TO_LEFT);
29                                   }
30                               });
31                           }
32                       }
33                   }
34               });
35   
36               // تهيئة الداتابيز
37               DatabaseManager_online.initializeDatabase();
38               DatabaseManager_online.createDefaultAdmin();
39   
40               // عرض شاشة الدخول
41               LoginScreen.show(stage);
42   
43               // تأمين تطبيق الاتجاه العربي
44               if (stage.getScene() != null && stage.getScene().getRoot() != null) {
45                   stage.getScene().getRoot().setNodeOrientation(NodeOrientation.RIGHT_TO_LEFT);
46               }
47   
48           } catch (Throwable t) {
49               t.printStackTrace();
50               // إظهار نافذة خطأ صريحة من الويندوز بدلاً من الإغلاق الصامت
51               JOptionPane.showMessageDialog(null,
52                       "حدث خطأ أثناء تشغيل التطبيق:\n" + t.toString() + "\n" + (t.getCause() != null ? t.getCause() : ""),
53                       "خطأ في التشغيل",
54                       JOptionPane.ERROR_MESSAGE);
55           }
56           // Check for updates in background
57           UpdateChecker.check();
58       }
59   
60       public static void main(String[] args) {
61           launch(args);
62           // بعد ما البرنامج يفتح
63       }
64   }
