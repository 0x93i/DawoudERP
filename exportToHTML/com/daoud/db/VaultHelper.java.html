8        public static int getWarehouseTreasuryId(int warehouseId) {
9            String sql = "SELECT id FROM vaults WHERE owner_type = 'warehouse' AND owner_id = ?";
10           try (Connection conn = DatabaseManager_online.getConnection();
11                PreparedStatement stmt = conn.prepareStatement(sql)) {
12               stmt.setInt(1, warehouseId);
13               ResultSet rs = stmt.executeQuery();
14               if (rs.next()) return rs.getInt("id");
15           } catch (SQLException e) {
16               System.err.println("VaultHelper error: " + e.getMessage());
17           }
18           return -1;
19       }

35       public static void record(int treasuryId, String direction, String paymentType,
36                                 double amount, String category, String notes, int recordedBy) {
37           if (treasuryId == -1 || amount <= 0) return;
38           String sql = "INSERT INTO vault_transactions " +
39                   "(vault_id, transaction_date, direction, payment_type, amount, category, notes, recorded_by) " +
40                   "VALUES (?, CURRENT_DATE, ?, ?, ?, ?, ?, ?)";
41           try (Connection conn = DatabaseManager_online.getConnection();
42                PreparedStatement stmt = conn.prepareStatement(sql)) {
43               stmt.setInt(1, treasuryId);
44               stmt.setString(2, direction); // 'in' or 'out'
45               stmt.setString(3, paymentType); // 'cash', 'bank', 'wallet', 'check'
46               stmt.setDouble(4, amount);
47               stmt.setString(5, category);
48               stmt.setString(6, notes);
49               stmt.setInt(7, recordedBy);
50               stmt.executeUpdate();
51           } catch (SQLException e) {
52               System.err.println("VaultHelper record error: " + e.getMessage());
53           }
54       }

56       // تحويل من نوع الدفع العربي لـ enum
57       public static String toPaymentType(String arabic) {
58           return switch (arabic) {
59               case "بنك", "تحويل بنكي" -> "bank";
60               case "محفظة" -> "wallet";
61               case "شيك" -> "check";
62               default -> "cash";
63           };
64       }
65       public static double getBalance(int vaultId, String paymentType) {
66           String sql = "SELECT COALESCE(SUM(CASE WHEN direction='in' THEN amount ELSE -amount END), 0) " +
67                   "FROM vault_transactions WHERE vault_id = ? AND payment_type = ?";
68           try (Connection conn = DatabaseManager_online.getConnection();
69                PreparedStatement stmt = conn.prepareStatement(sql)) {
70               stmt.setInt(1, vaultId);
71               stmt.setString(2, paymentType);
72               ResultSet rs = stmt.executeQuery();
73               if (rs.next()) return rs.getDouble(1);
74           } catch (SQLException e) {
75               System.err.println("Balance check error: " + e.getMessage());
76           }
77           return 0;
78       }
