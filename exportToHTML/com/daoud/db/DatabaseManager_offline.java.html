331      public static void createDefaultAdmin() {
332          String check = "SELECT COUNT(*) FROM users WHERE username = 'admin'";
333          String insert = "INSERT INTO users (username, password_hash, role) VALUES ('admin', 'admin123', 'admin')";
334          try (Connection conn = getConnection();
335               PreparedStatement checkStmt = conn.prepareStatement(check)) {
336              ResultSet rs = checkStmt.executeQuery();
337              if (rs.next() && rs.getInt(1) == 0) {
338                  conn.createStatement().execute(insert);
339                  System.out.println("Default admin created.");
340              }
341          } catch (SQLException e) {
342              System.err.println("Error creating admin: " + e.getMessage());
343          }
344      }
