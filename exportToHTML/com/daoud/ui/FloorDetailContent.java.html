16       public static Node build(int userId, String username, String role, int supplierId, String supplierName) {
17   
18           Label balanceLabel = new Label("جاري التحميل...");
19           balanceLabel.setStyle("-fx-font-size: 22px; -fx-font-weight: bold;");
20   
21           List<FloorDAO.FloorTransaction> allRows = new java.util.ArrayList<>();
22           VBox tableBody = new VBox(0);
23           tableBody.getChildren().add(new Label("جاري التحميل..."));
24   
25           Button backBtn = new Button("← رجوع للأرضيات"); backBtn.getStyleClass().add("btn-default");
26           backBtn.setOnAction(e -> {
27               MainLayout.loadContent(FloorsContent.build(userId, username, role));
28               MainLayout.setTitle("الأرضيات");
29           });
30   
31           Button openAccountBtn = new Button("فتح حساب المورد"); openAccountBtn.getStyleClass().add("btn-default");
32           openAccountBtn.setOnAction(e -> {
33               // نجيب بيانات المورد كاملة عشان نفتح صفحته العادية
34               AsyncHelper.run(
35                       () -> SupplierDAO.getAllSuppliers().stream()
36                               .filter(s -> s.getId() == supplierId).findFirst().orElse(null),
37                       supplier -> {
38                           if (supplier != null) {
39                               MainLayout.loadContent(SupplierDetailContent.build(userId, username, role, supplier));
40                               MainLayout.setTitle("حساب مورد: " + supplier.getName());
41                           }
42                       },
43                       error -> {}
44               );
45           });
46   
47           VBox infoCard = new VBox(8); infoCard.getStyleClass().add("card");
48           Label title = new Label("أرضية: " + supplierName); title.getStyleClass().add("card-title");
49           infoCard.getChildren().addAll(title, balanceLabel, new HBox(8, openAccountBtn));
50   
51           // ── فورم زيادة/نقصان (أدمن بس) ──
52           VBox adjustCard = new VBox(10); adjustCard.getStyleClass().add("card");
53           adjustCard.setVisible(role.equals("admin")); adjustCard.setManaged(role.equals("admin"));
54   
55           ComboBox<String> directionCombo = new ComboBox<>(FXCollections.observableArrayList("زيادة", "نقصان"));
56           directionCombo.setValue("زيادة"); directionCombo.setMaxWidth(Double.MAX_VALUE);
57           TextField amountField = new TextField(); amountField.setPromptText("المبلغ"); amountField.setMaxWidth(Double.MAX_VALUE);
58           TextField notesField = new TextField(); notesField.setPromptText("ملاحظة (اختياري)"); notesField.setMaxWidth(Double.MAX_VALUE);
59           Button saveBtn = new Button("تسجيل"); saveBtn.getStyleClass().add("btn-primary");
60           Label saveMsg = new Label("");
61   
62           saveBtn.setOnAction(e -> {
63               double amount;
64               try {
65                   amount = Double.parseDouble(amountField.getText().trim());
66               } catch (NumberFormatException ex) { saveMsg.setText("ادخل رقم صحيح"); return; }
67               if (amount <= 0) { saveMsg.setText("ادخل رقم أكبر من صفر"); return; }
68   
69               String direction = directionCombo.getValue().equals("زيادة") ? "in" : "out";
70               String notes = notesField.getText().trim();
71   
72               saveMsg.setText("جاري الحفظ...");
73               AsyncHelper.runVoid(
74                       () -> FloorDAO.adjust(supplierId, direction, amount, notes, userId),
75                       () -> {
76                           amountField.clear(); notesField.clear();
77                           saveMsg.setText("تم ✓");
78                           reload(balanceLabel, tableBody, allRows, supplierId);
79                       },
80                       error -> saveMsg.setText("حصل خطأ أثناء الحفظ"),
81                       saveBtn
82               );
83           });
84   
85           Label adjustTitle = new Label("تعديل رصيد الأرضية"); adjustTitle.getStyleClass().add("card-title");
86           adjustCard.getChildren().addAll(adjustTitle,
87                   new HBox(8,
88                           new VBox(4, new Label("النوع:"), directionCombo),
89                           new VBox(4, new Label("المبلغ:"), amountField)),
90                   new VBox(4, new Label("ملاحظة:"), notesField),
91                   saveBtn, saveMsg);
92   
93           VBox historyCard = new VBox(8); historyCard.getStyleClass().add("card");
94           Label histTitle = new Label("سجل حركات الأرضية"); histTitle.getStyleClass().add("card-title");
95           historyCard.getChildren().addAll(histTitle, buildTableHeader(), tableBody);
96   
97           reload(balanceLabel, tableBody, allRows, supplierId);
98   
99           return new VBox(12, backBtn, infoCard, adjustCard, historyCard);
100      }

102      private static HBox buildTableHeader() {
103          HBox header = new HBox();
104          header.setStyle("-fx-background-color: #f5f5f3; -fx-padding: 8 10;");
105          String[] cols = {"التاريخ", "النوع", "المبلغ", "ملاحظة", "بواسطة"};
106          double[] widths = {95, 80, 100, 200, 120};
107          for (int i = 0; i < cols.length; i++) {
108              Label lbl = new Label(cols[i]);
109              lbl.setStyle("-fx-font-size: 11px; -fx-font-weight: bold; -fx-text-fill: #888780;");
110              lbl.setMinWidth(widths[i]); lbl.setPrefWidth(widths[i]);
111              header.getChildren().add(lbl);
112          }
113          return header;
114      }

116      private static void reload(Label balanceLabel, VBox tableBody, List<FloorDAO.FloorTransaction> allRows, int supplierId) {
117          AsyncHelper.run(
118                  () -> new Object[]{FloorDAO.getFloorAmount(supplierId), FloorDAO.getHistory(supplierId)},
119                  result -> {
120                      double balance = (double) result[0];
121                      @SuppressWarnings("unchecked")
122                      List<FloorDAO.FloorTransaction> rows = (List<FloorDAO.FloorTransaction>) result[1];
123                      balanceLabel.setText(String.format("%.0f جنيه", balance));
124                      balanceLabel.setStyle("-fx-font-size: 22px; -fx-font-weight: bold; -fx-text-fill: " +
125                              (balance >= 0 ? "#3B6D11" : "#A32D2D") + ";");
126                      allRows.clear(); allRows.addAll(rows);
127                      renderTable(tableBody, allRows);
128                  },
129                  error -> balanceLabel.setText("حصل خطأ")
130          );
131      }
132  
133      private static void renderTable(VBox table, List<FloorDAO.FloorTransaction> rows) {
134          table.getChildren().clear();
135          double[] widths = {95, 80, 100, 200, 120};
136          boolean odd = true;
137  
138          if (rows.isEmpty()) {
139              Label empty = new Label("مفيش حركات لسه");
140              empty.setStyle("-fx-text-fill: #888780; -fx-padding: 16; -fx-font-size: 13px;");
141              table.getChildren().add(empty);
142              return;
143          }
144  
145          for (FloorDAO.FloorTransaction row : rows) {
146              HBox r = new HBox(4);
147              r.setStyle("-fx-background-color: " + (odd ? "#ffffff" : "#fafaf8") +
148                      "; -fx-padding: 7 10; -fx-border-color: transparent transparent #f0f0f0 transparent;");
149              odd = !odd;
150  
151              boolean isIn = row.direction().equals("in");
152  
153              Label dateLbl = new Label(row.date());
154              dateLbl.setStyle("-fx-font-size: 12px; -fx-text-fill: #5f5e5a;");
155              dateLbl.setMinWidth(widths[0]); dateLbl.setPrefWidth(widths[0]);
156  
157              Label dirLbl = new Label(isIn ? "زيادة" : "نقصان");
158              dirLbl.setStyle("-fx-font-size: 11px; -fx-padding: 2 8; -fx-background-radius: 4; -fx-background-color: " +
159                      (isIn ? "#EAF3DE; -fx-text-fill: #3B6D11" : "#FCEBEB; -fx-text-fill: #A32D2D") + ";");
160              dirLbl.setMinWidth(widths[1]); dirLbl.setPrefWidth(widths[1]);
161  
162              Label amtLbl = new Label(String.format("%.0f جنيه", row.amount()));
163              amtLbl.setStyle("-fx-font-size: 12px; -fx-font-weight: bold; -fx-text-fill: " + (isIn ? "#3B6D11" : "#A32D2D") + ";");
164              amtLbl.setMinWidth(widths[2]); amtLbl.setPrefWidth(widths[2]);
165  
166              Label notesLbl = new Label(row.notes());
167              notesLbl.setStyle("-fx-font-size: 12px; -fx-text-fill: #888780;");
168              notesLbl.setMinWidth(widths[3]); notesLbl.setPrefWidth(widths[3]);
169  
170              Label byLbl = new Label(row.recordedByName());
171              byLbl.setStyle("-fx-font-size: 12px; -fx-text-fill: #888780;");
172              byLbl.setMinWidth(widths[4]); byLbl.setPrefWidth(widths[4]);
173  
174              r.getChildren().addAll(dateLbl, dirLbl, amtLbl, notesLbl, byLbl);
175              table.getChildren().add(r);
176          }
177      }
