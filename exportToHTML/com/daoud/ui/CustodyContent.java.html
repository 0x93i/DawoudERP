CustodyContent.java 
1    package com.daoud.ui;
2    
3    import com.daoud.dao.CustodyDAO;
4    import com.daoud.dao.WarehouseDAO;
5    import com.daoud.model.Supplier;
6    import com.daoud.model.Warehouse;
7    import javafx.collections.FXCollections;
8    import javafx.geometry.Insets;
9    import javafx.scene.Node;
10   import javafx.scene.control.*;
11   import javafx.scene.layout.*;
12   
13   import java.util.List;
14   
15   public class CustodyContent {
16   
17       public static Node build(int userId, String username, String role) {
18   
19           TextArea custodyHistory = new TextArea();
20           custodyHistory.setEditable(false);
21           custodyHistory.setPrefHeight(120);
22   
23           TextArea disburseHistory = new TextArea();
24           disburseHistory.setEditable(false);
25           disburseHistory.setPrefHeight(120);
26   
27           List<Warehouse> warehouses = WarehouseDAO.getAllWarehouses();
28           ComboBox<Warehouse> warehouseCombo = new ComboBox<>(FXCollections.observableArrayList(warehouses));
29           warehouseCombo.setPromptText("اختار المخزن");
30   
31           Label remainingLabel = new Label("المتبقي: —");
32   
33           // ── إضافة عهدة (أدمن بس) ──
34           TextField custodyAmount = new TextField(); custodyAmount.setPromptText("المبلغ");
35           TextField custodyNotes = new TextField(); custodyNotes.setPromptText("ملاحظة");
36           Button addCustodyBtn = new Button("إضافة عهدة"); addCustodyBtn.getStyleClass().add("btn-primary");
37           Label custodyMsg = new Label("");
38   
39           custodyAmount.setVisible(role.equals("admin")); custodyAmount.setManaged(role.equals("admin"));
40           custodyNotes.setVisible(role.equals("admin")); custodyNotes.setManaged(role.equals("admin"));
41           addCustodyBtn.setVisible(role.equals("admin")); addCustodyBtn.setManaged(role.equals("admin"));
42           custodyMsg.setVisible(role.equals("admin")); custodyMsg.setManaged(role.equals("admin"));
43   
44           // ── توزيع ──
45           ComboBox<String> recipientTypeCombo = new ComboBox<>(FXCollections.observableArrayList("مورد", "أخرى"));
46           recipientTypeCombo.setPromptText("نوع المستلم");
47   
48           ComboBox<Supplier> supplierCombo = new ComboBox<>();
49           supplierCombo.setPromptText("اختار المورد");
50           supplierCombo.setVisible(false); supplierCombo.setManaged(false);
51   
52           TextField otherNotes = new TextField(); otherNotes.setPromptText("وصف");
53           otherNotes.setVisible(false); otherNotes.setManaged(false);
54   
55           TextField disburseAmount = new TextField(); disburseAmount.setPromptText("المبلغ");
56           Button disburseBtn = new Button("تسجيل التوزيع"); disburseBtn.getStyleClass().add("btn-default");
57           Label disburseMsg = new Label("");
58   
59           recipientTypeCombo.setOnAction(e -> {
60               String type = recipientTypeCombo.getValue();
61               supplierCombo.setVisible("مورد".equals(type)); supplierCombo.setManaged("مورد".equals(type));
62               otherNotes.setVisible("أخرى".equals(type)); otherNotes.setManaged("أخرى".equals(type));
63           });
64   
65           warehouseCombo.setOnAction(e -> {
66               Warehouse w = warehouseCombo.getValue();
67               if (w == null) return;
68               remainingLabel.setText(String.format("المتبقي: %.1f جنيه", CustodyDAO.getRemainingCustody(w.getId())));
69               supplierCombo.setItems(FXCollections.observableArrayList(WarehouseDAO.getSuppliersByWarehouse(w.getId())));
70               custodyHistory.setText(String.join("\n", CustodyDAO.getCustodyHistory(w.getId())));
71               disburseHistory.setText(String.join("\n", CustodyDAO.getDisbursementHistory(w.getId())));
72           });
73   
74           addCustodyBtn.setOnAction(e -> {
75               Warehouse w = warehouseCombo.getValue();
76               if (w == null) { custodyMsg.setText("اختار مخزن"); return; }
77               try {
78                   double amount = Double.parseDouble(custodyAmount.getText().trim());
79                   CustodyDAO.addCustody(w.getId(), userId, amount, custodyNotes.getText().trim());
80                   remainingLabel.setText(String.format("المتبقي: %.1f جنيه", CustodyDAO.getRemainingCustody(w.getId())));
81                   custodyAmount.clear(); custodyNotes.clear();
82                   custodyMsg.setText("تم ✓");
83                   custodyHistory.setText(String.join("\n", CustodyDAO.getCustodyHistory(w.getId())));
84               } catch (NumberFormatException ex) { custodyMsg.setText("ادخل رقم"); }
85           });
86   
87           disburseBtn.setOnAction(e -> {
88               Warehouse w = warehouseCombo.getValue();
89               if (w == null) { disburseMsg.setText("اختار مخزن"); return; }
90               if (recipientTypeCombo.getValue() == null) { disburseMsg.setText("اختار نوع المستلم"); return; }
91               try {
92                   double amount = Double.parseDouble(disburseAmount.getText().trim());
93                   double remaining = CustodyDAO.getRemainingCustody(w.getId());
94                   if (amount > remaining) { disburseMsg.setText("المبلغ أكبر من المتبقي"); return; }
95                   String type = recipientTypeCombo.getValue().equals("مورد") ? "supplier" : "other";
96                   int recipientId = 0;
97                   String notes = "";
98                   if (type.equals("supplier")) {
99                       Supplier s = supplierCombo.getValue();
100                      if (s == null) { disburseMsg.setText("اختار المورد"); return; }
101                      recipientId = s.getId(); notes = s.getName();
102                  } else { notes = otherNotes.getText().trim(); }
103                  CustodyDAO.addDisbursement(w.getId(), userId, type, recipientId, amount, notes);
104                  remainingLabel.setText(String.format("المتبقي: %.1f جنيه", CustodyDAO.getRemainingCustody(w.getId())));
105                  disburseAmount.clear(); disburseMsg.setText("تم ✓");
106                  disburseHistory.setText(String.join("\n", CustodyDAO.getDisbursementHistory(w.getId())));
107              } catch (NumberFormatException ex) { disburseMsg.setText("ادخل رقم"); }
108          });
109  
110          // ── Cards ──
111          VBox warehouseCard = new VBox(8); warehouseCard.getStyleClass().add("card");
112          warehouseCard.getChildren().addAll(new Label("المخزن:"), warehouseCombo, remainingLabel);
113  
114          VBox custodyCard = new VBox(8); custodyCard.getStyleClass().add("card");
115          Label custodyTitle = new Label("إضافة عهدة (أدمن فقط)");
116          custodyTitle.setVisible(role.equals("admin")); custodyTitle.setManaged(role.equals("admin"));
117          custodyCard.getChildren().addAll(custodyTitle, new Label("المبلغ:"), custodyAmount, new Label("ملاحظة:"), custodyNotes, addCustodyBtn, custodyMsg);
118  
119          VBox disburseCard = new VBox(8); disburseCard.getStyleClass().add("card");
120          disburseCard.getChildren().addAll(
121                  new Label("تسجيل توزيع:"),
122                  new Label("نوع المستلم:"), recipientTypeCombo,
123                  supplierCombo, otherNotes,
124                  new Label("المبلغ:"), disburseAmount,
125                  disburseBtn, disburseMsg);
126  
127          VBox historyCard = new VBox(8); historyCard.getStyleClass().add("card");
128          historyCard.getChildren().addAll(
129                  new Label("سجل العهد:"), custodyHistory,
130                  new Label("سجل التوزيع:"), disburseHistory);
131  
132          return new VBox(12, warehouseCard, custodyCard, disburseCard, historyCard);
133      }
134  }
