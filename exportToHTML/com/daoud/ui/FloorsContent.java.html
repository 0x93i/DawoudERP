19       public static Node build(int userId, String username, String role) {
20   
21           VBox totalsRow = new VBox(0);
22           VBox tableBody = new VBox(0);
23           tableBody.getChildren().add(new Label("جاري التحميل..."));
24   
25           HBox filtersBox = new HBox(6);
26           filtersBox.setAlignment(Pos.CENTER_RIGHT);
27   
28           List<FloorDAO.FloorRow> allFloors = new ArrayList<>();
29           List<Warehouse> warehouses = new ArrayList<>();
30   
31           AsyncHelper.run(
32                   () -> new FloorsData(FloorDAO.getAllFloors(), WarehouseDAO.getAllWarehouses()),
33                   data -> {
34                       allFloors.clear(); allFloors.addAll(data.floors());
35                       warehouses.clear(); warehouses.addAll(data.warehouses());
36                       buildFilters(filtersBox, allFloors, warehouses, totalsRow, tableBody, userId, username, role);
37                       applyFilter(null, allFloors, warehouses, totalsRow, tableBody, userId, username, role);
38                   },
39                   error -> tableBody.getChildren().setAll(new Label("حصل خطأ في تحميل الأرضيات"))
40           );
41   
42           VBox card = new VBox(12);
43           card.getStyleClass().add("card");
44           Label title = new Label("الأرضيات"); title.getStyleClass().add("card-title");
45           card.getChildren().addAll(title, filtersBox, totalsRow, buildHeader(), tableBody);
46   
47           return new VBox(12, card);
48       }

50       // filter: null = الكل، -1 = عام، أو id المخزن
51       private static void buildFilters(HBox filtersBox, List<FloorDAO.FloorRow> allFloors, List<Warehouse> warehouses,
52                                        VBox totalsRow, VBox tableBody, int userId, String username, String role) {
53           filtersBox.getChildren().clear();
54   
55           List<Button> buttons = new ArrayList<>();
56           Button allBtn = filterBtn("الكل");
57           Button generalBtn = filterBtn("موردين عام");
58           buttons.add(allBtn); buttons.add(generalBtn);
59   
60           allBtn.setOnAction(e -> {
61               setActive(allBtn, buttons);
62               applyFilter(null, allFloors, warehouses, totalsRow, tableBody, userId, username, role);
63           });
64           generalBtn.setOnAction(e -> {
65               setActive(generalBtn, buttons);
66               applyFilter(-1, allFloors, warehouses, totalsRow, tableBody, userId, username, role);
67           });
68   
69           for (Warehouse w : warehouses) {
70               Button wBtn = filterBtn(w.getName());
71               buttons.add(wBtn);
72               wBtn.setOnAction(e -> {
73                   setActive(wBtn, buttons);
74                   applyFilter(w.getId(), allFloors, warehouses, totalsRow, tableBody, userId, username, role);
75               });
76           }
77   
78           filtersBox.getChildren().addAll(buttons);
79           setActive(allBtn, buttons);
80       }
81   
82       private static void applyFilter(Integer warehouseFilter, List<FloorDAO.FloorRow> allFloors, List<Warehouse> warehouses,
83                                       VBox totalsRow, VBox tableBody, int userId, String username, String role) {
84           List<FloorDAO.FloorRow> filtered = new ArrayList<>();
85           for (FloorDAO.FloorRow f : allFloors) {
86               if (warehouseFilter == null || f.warehouseId() == warehouseFilter) filtered.add(f);
87           }
88           renderTotals(totalsRow, filtered, warehouseFilter, warehouses);
89           renderTable(tableBody, filtered, userId, username, role);
90       }
91   
92       private static void renderTotals(VBox totalsRow, List<FloorDAO.FloorRow> rows, Integer warehouseFilter, List<Warehouse> warehouses) {
93           totalsRow.getChildren().clear();
94   
95           double total = 0;
96           for (FloorDAO.FloorRow f : rows) total += f.floorAmount();
97   
98           String label;
99           if (warehouseFilter == null) label = "إجمالي أرضيات كل الموردين";
100          else if (warehouseFilter == -1) label = "إجمالي أرضيات الموردين العام";
101          else {
102              String name = "المخزن";
103              for (Warehouse w : warehouses) if (w.getId() == warehouseFilter) { name = w.getName(); break; }
104              label = "إجمالي أرضيات موردي: " + name;
105          }
106  
107          Label nameLbl = new Label(label);
108          nameLbl.setStyle("-fx-font-size: 12px; -fx-text-fill: #888780;");
109          Label totalLbl = new Label(String.format("%.0f جنيه", total));
110          totalLbl.setStyle("-fx-font-size: 18px; -fx-font-weight: bold; -fx-text-fill: " +
111                  (total >= 0 ? "#3B6D11" : "#A32D2D") + ";");
112  
113          VBox box = new VBox(2, nameLbl, totalLbl);
114          box.setStyle("-fx-background-color: #f5f5f3; -fx-padding: 10 14; -fx-background-radius: 8;");
115          totalsRow.getChildren().add(box);
116      }

118      private static Button filterBtn(String text) {
119          Button btn = new Button(text);
120          btn.setStyle("-fx-font-size: 11px; -fx-padding: 4 10; -fx-border-radius: 4; -fx-background-radius: 4; -fx-cursor: hand; -fx-background-color: white; -fx-border-color: #c0c0c0;");

124      private static void setActive(Button active, List<Button> all) {
125          for (Button b : all) {
126              if (b == active)
127                  b.setStyle("-fx-font-size: 11px; -fx-padding: 4 10; -fx-border-radius: 4; -fx-background-radius: 4; -fx-cursor: hand; -fx-background-color: #3B6D11; -fx-text-fill: white;");
128              else
129                  b.setStyle("-fx-font-size: 11px; -fx-padding: 4 10; -fx-border-radius: 4; -fx-background-radius: 4; -fx-cursor: hand; -fx-background-color: white; -fx-border-color: #c0c0c0; -fx-text-fill: #1a1a18;");
130          }
131      }

133      private static HBox buildHeader() {
134          HBox header = new HBox();
135          header.setStyle("-fx-background-color: #f5f5f3; -fx-padding: 8 10; -fx-background-radius: 6;");
136          String[] cols = {"رقم", "المورد", "رصيد الأرضية", ""};
137          double[] widths = {50, 260, 150, 100};
138          for (int i = 0; i < cols.length; i++) {
139              Label lbl = new Label(cols[i]);
140              lbl.setStyle("-fx-font-size: 11px; -fx-font-weight: bold; -fx-text-fill: #888780;");
141              lbl.setMinWidth(widths[i]); lbl.setPrefWidth(widths[i]);
142              header.getChildren().add(lbl);
143          }
144          return header;
145      }

147      private static void renderTable(VBox table, List<FloorDAO.FloorRow> floors, int userId, String username, String role) {
148          table.getChildren().clear();
149  
150          if (floors.isEmpty()) {
151              Label empty = new Label("مفيش موردين");
152              empty.setStyle("-fx-text-fill: #888780; -fx-padding: 20; -fx-font-size: 13px;");
153              table.getChildren().add(empty);
154              return;
155          }
156  
157          boolean odd = true;
158          for (FloorDAO.FloorRow f : floors) {
159              HBox row = new HBox(4);
160              row.setStyle("-fx-background-color: " + (odd ? "#ffffff" : "#fafaf8") +
161                      "; -fx-padding: 9 10; -fx-border-color: transparent transparent #f0f0f0 transparent;");
162              odd = !odd;
163  
164              Label numLbl = new Label(f.supplierNo() > 0 ? String.valueOf(f.supplierNo()) : "—");
165              numLbl.setStyle("-fx-font-size: 12px; -fx-font-weight: bold; -fx-text-fill: #5f5e5a;");
166              numLbl.setMinWidth(50); numLbl.setPrefWidth(50);
167  
168              Label nameLbl = new Label(f.name());
169              nameLbl.setStyle("-fx-font-size: 13px; -fx-text-fill: #1a1a18;");
170              nameLbl.setMinWidth(260); nameLbl.setPrefWidth(260);
171  
172              Label amountLbl = new Label(String.format("%.0f جنيه", f.floorAmount()));
173              amountLbl.setStyle("-fx-font-size: 13px; -fx-font-weight: bold; -fx-text-fill: " +
174                      (f.floorAmount() >= 0 ? "#3B6D11" : "#A32D2D") + ";");
175              amountLbl.setMinWidth(150); amountLbl.setPrefWidth(150);
176  
177              Button openBtn = new Button("فتح");
178              openBtn.setStyle("-fx-font-size: 11px; -fx-padding: 4 14; -fx-background-color: white; -fx-border-color: #c0c0c0; -fx-border-radius: 4; -fx-background-radius: 4; -fx-cursor: hand;");
179              openBtn.setMinWidth(100); openBtn.setPrefWidth(100);
180              openBtn.setOnAction(e -> {
181                  MainLayout.loadContent(FloorDetailContent.build(userId, username, role, f.supplierId(), f.name()));
182                  MainLayout.setTitle("أرضية: " + f.name());
183              });
184  
185              row.getChildren().addAll(numLbl, nameLbl, amountLbl, openBtn);
186              table.getChildren().add(row);
187          }
188      }
