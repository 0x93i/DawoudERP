1    package com.daoud.ui;
2    
3    import com.daoud.dao.FactoryDAO;
4    import com.daoud.model.Factory;
5    import javafx.collections.FXCollections;
6    import javafx.geometry.Insets;
7    import javafx.geometry.Pos;
8    import javafx.scene.Node;
9    import javafx.scene.control.*;
10   import javafx.scene.layout.*;
11   import javafx.stage.Stage;
12   
13   public class FactoriesContent {
14   
15       public static Node build(int userId, String username, String role) {
16           ListView<Factory> listView = new ListView<>();
17           listView.getStyleClass().add("list-view");
18           listView.setPrefHeight(400);
19           listView.setPlaceholder(new Label("جاري تحميل المصانع..."));
20           refreshList(listView);
21   
22           Button addBtn = new Button("إضافة مصنع");
23           addBtn.getStyleClass().add("btn-primary");
24           addBtn.setDisable(!role.equals("admin"));
25   
26           Button openBtn = new Button("فتح الحساب");
27           openBtn.getStyleClass().add("btn-default");
28           openBtn.setDisable(true);
29   
30           Button deleteBtn = new Button("حذف");
31           deleteBtn.getStyleClass().add("btn-danger");
32           deleteBtn.setDisable(true);
33   
34           listView.getSelectionModel().selectedItemProperty().addListener((obs, old, selected) -> {
35               openBtn.setDisable(selected == null);
36               deleteBtn.setDisable(selected == null || !role.equals("admin"));
37           });
38   
39           addBtn.setOnAction(e -> {
40               TextField nameField = new TextField();
41               nameField.setPromptText("اسم المصنع");
42               Button saveBtn = new Button("حفظ");
43               saveBtn.getStyleClass().add("btn-primary");
44               Label errLbl = new Label("");
45               VBox layout = new VBox(10, new Label("الاسم:"), nameField, saveBtn, errLbl);
46               layout.setPadding(new Insets(20));
47               Stage dialog = DialogHelper.create("إضافة مصنع", layout, 320, 180);
48               saveBtn.setOnAction(ev -> {
49                   if (nameField.getText().trim().isEmpty()) { errLbl.setText("الاسم مطلوب"); return; }
50                   String name = nameField.getText().trim();
51                   errLbl.setStyle("-fx-text-fill: #5f5e5a;");
52                   errLbl.setText("جاري الحفظ...");
53                   AsyncHelper.runVoid(
54                           () -> FactoryDAO.addFactory(name, userId),
55                           () -> { refreshList(listView); dialog.close(); },
56                           error -> { errLbl.setStyle("-fx-text-fill: #b91c1c;"); errLbl.setText("حصل خطأ"); },
57                           saveBtn
58                   );
59               });
60               dialog.show();
61           });
62   
63           deleteBtn.setOnAction(e -> {
64               Factory selected = listView.getSelectionModel().getSelectedItem();
65               if (selected != null) {
66                   Alert confirm = new Alert(Alert.AlertType.CONFIRMATION, "هتحذف: " + selected.getName() + "؟");
67                   confirm.showAndWait().ifPresent(r -> {
68                       if (r == ButtonType.OK) {
69                           deleteBtn.setDisable(true);
70                           AsyncHelper.runVoid(
71                                   () -> FactoryDAO.deleteFactory(selected.getId()),
72                                   () -> refreshList(listView),
73                                   error -> refreshList(listView)
74                           );
75                       }
76                   });
77               }
78           });
79   
80           openBtn.setOnAction(e -> {
81               Factory selected = listView.getSelectionModel().getSelectedItem();
82               if (selected != null) {
83                   MainLayout.loadContent(FactoryDetailContent.build(userId, username, role, selected));
84                   MainLayout.setTitle("حساب مصنع: " + selected.getName());
85               }
86           });
87   
88           HBox buttons = new HBox(10, addBtn, openBtn, deleteBtn);
89           buttons.setAlignment(Pos.CENTER_LEFT);
90   
91           VBox card = new VBox(10);
92           card.getStyleClass().add("card");
93           Label cardTitle = new Label("قائمة المصانع");
94           cardTitle.getStyleClass().add("card-title");
95           card.getChildren().addAll(cardTitle, buttons, listView);
96   
97           return new VBox(12, card);
98       }
99   
100      private static void refreshList(ListView<Factory> listView) {
101          AsyncHelper.run(
102                  FactoryDAO::getAllFactories,
103                  factories -> listView.setItems(FXCollections.observableArrayList(factories)),
104                  error -> listView.setPlaceholder(new Label("حصل خطأ في تحميل المصانع"))
105          );
106      }
107  }
