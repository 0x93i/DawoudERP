232      private static List<HistoryRow> loadRows(int factoryId) {
233          List<HistoryRow> rows = new ArrayList<>();
234          // شحنات
235          String sql1 = "SELECT id, shipment_date, gross_weight, deduction_pct, deduction_kg, price_per_kg, total_amount, COALESCE(supplier_name,'') as supplier_name " +
236                  "FROM factory_shipments WHERE factory_id = ? ORDER BY shipment_date DESC LIMIT 30";
237  
238          try (Connection conn = DatabaseManager_online.getConnection();
239               PreparedStatement stmt = conn.prepareStatement(sql1)) {
240              stmt.setInt(1, factoryId);
241              ResultSet rs = stmt.executeQuery();
242              while (rs.next()) {
243                  rows.add(new HistoryRow("شحنة", rs.getInt("id"), rs.getString("shipment_date"),
244                          "—",
245                          rs.getString("supplier_name") != null && !rs.getString("supplier_name").isEmpty() ? rs.getString("supplier_name") : "—",
246                          String.format("%.1f كيلو", rs.getDouble("gross_weight")),
247                          String.format("%.1f%%", rs.getDouble("deduction_pct")),
248                          String.format("%.1f كيلو", rs.getDouble("deduction_kg")),
249                          String.format("%.2f جنيه", rs.getDouble("price_per_kg")),
250                          String.format("%.0f جنيه", rs.getDouble("total_amount"))));
251              }
252          } catch (SQLException e) { System.err.println(e.getMessage()); }
253  
254          // دفعات
255          String sql2 = "SELECT id, payment_date, amount, COALESCE(notes,'') as notes " +
256                  "FROM factory_payments WHERE factory_id = ? ORDER BY payment_date DESC LIMIT 30";
257          try (Connection conn = DatabaseManager_online.getConnection();
258               PreparedStatement stmt = conn.prepareStatement(sql2)) {
259              stmt.setInt(1, factoryId);
260              ResultSet rs = stmt.executeQuery();
261              while (rs.next()) {
262                  String notes = rs.getString("notes");
263                  rows.add(new HistoryRow("دفعة", rs.getInt("id"), rs.getString("payment_date"),
264                          notes != null && !notes.isEmpty() ? notes : "—",
265                          "—", "—", "—", "—", "—",
266                          String.format("%.0f جنيه", rs.getDouble("amount"))));
267              }
268          } catch (SQLException e) { System.err.println(e.getMessage()); }
269  
270          rows.sort((a, b) -> b.date().compareTo(a.date()));
271          return rows;
272      }
