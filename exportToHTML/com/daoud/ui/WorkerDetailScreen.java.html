121      private static void refreshSummary(Label lbl, int workerId) {
122          int days = WorkerDAO.getWorkDays(workerId);
123          double earned = WorkerDAO.getTotalEarned(workerId);
124          double withdrawn = WorkerDAO.getTotalWithdrawn(workerId);
125          double remaining = earned - withdrawn;
126          lbl.setText(String.format("أيام العمل: %d | إجمالي الأجر: %.1f | المسحوب: %.1f | المتبقي: %.1f جنيه",
127                  days, earned, withdrawn, remaining));
128      }

130      private static void refreshHistory(TextArea area, int workerId) {
131          StringBuilder sb = new StringBuilder();
132          String sql = """ 
133              SELECT 'حضور' as type, work_date as date, daily_wage as amount, '' as notes 
134              FROM worker_attendance WHERE worker_id = ? 
135              UNION ALL 
136              SELECT 'سحب' as type, withdrawal_date as date, amount, COALESCE(notes,'') 
137              FROM worker_withdrawals WHERE worker_id = ? 
138              ORDER BY date DESC LIMIT 30 
139          """;
140          try (Connection conn = DatabaseManager_online.getConnection();
141               PreparedStatement stmt = conn.prepareStatement(sql)) {
142              stmt.setInt(1, workerId);
143              stmt.setInt(2, workerId);
144              ResultSet rs = stmt.executeQuery();
145              while (rs.next()) {
146                  sb.append(String.format("%s | %s | %.1f جنيه %s%n",
147                          rs.getString("date"),
148                          rs.getString("type"),
149                          rs.getDouble("amount"),
150                          rs.getString("notes").isEmpty() ? "" : "| " + rs.getString("notes")));
151              }
152          } catch (SQLException e) {
153              sb.append("خطأ في تحميل السجل");
154          }
155          area.setText(sb.toString());
156      }
