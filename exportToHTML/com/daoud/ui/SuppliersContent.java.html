127      private static void loadSupplierRows(List<SupplierRow> rows, int userId, String role) {
128          rows.clear();
129          if (role.equals("admin")) {
130              // استعلام واحد بيرجع كل الموردين + اسم مخزن كل واحد فيهم مع بعض
131              // (بدل ما كان بيعمل استعلام منفصل لكل مورد لوحده - ده كان أبطأ حاجة في الشاشة)
132              rows.addAll(loadAllSupplierRowsWithWarehouse());
133          } else {
134              // مسؤول المخزن أصلاً بيشوف موردين مخزنه بس، يعني اسم المخزن معروف من غير أي استعلام إضافي
135              Warehouse w = WarehouseDAO.getWarehouseByManager(userId);
136              if (w != null) {
137                  List<Supplier> suppliers = WarehouseDAO.getSuppliersByWarehouse(w.getId());
138                  for (Supplier s : suppliers) rows.add(new SupplierRow(s, w.getName()));
139              }
140          }
141      }

143      private static List<SupplierRow> loadAllSupplierRowsWithWarehouse() {
144          List<SupplierRow> rows = new ArrayList<>();
145          String sql = """ 
146              SELECT s.id, s.name, s.phone, s.sector, s.floor_amount, s.floor_date, s.supplier_no, 
147                     (SELECT w.name FROM warehouses w 
148                      JOIN warehouse_suppliers ws ON w.id = ws.warehouse_id 
149                      WHERE ws.supplier_id = s.id LIMIT 1) AS warehouse_name 
150              FROM suppliers s 
151              ORDER BY s.supplier_no NULLS LAST, s.name 
152          """;
153          try (Connection conn = DatabaseManager_online.getConnection();
154               Statement stmt = conn.createStatement();
155               ResultSet rs = stmt.executeQuery(sql)) {
156              while (rs.next()) {
157                  Supplier s = new Supplier(
158                          rs.getInt("id"),
159                          rs.getString("name"),
160                          rs.getString("phone"),
161                          rs.getString("sector"),
162                          rs.getDouble("floor_amount"),
163                          rs.getString("floor_date"),
164                          rs.getInt("supplier_no")
165                  );
166                  rows.add(new SupplierRow(s, rs.getString("warehouse_name")));
167              }
168          } catch (SQLException e) {
169              System.err.println("Error: " + e.getMessage());
170          }
171          return rows;
172      }

188      private static void renderTable(VBox table, List<SupplierRow> rows, String filter,
189                                      int userId, String username, String role) {
190          table.getChildren().clear();
191          double[] widths = {50, 150, 110, 110, 90, 130, 110, 70, 70};
192          boolean odd = true, hasRows = false;
193  
194          for (SupplierRow row : rows) {
195              boolean isWarehouse = row.warehouseName() != null;
196  
197              if (filter != null) {
198                  if (filter.equals("warehouse") && !isWarehouse) continue;
199                  if (filter.equals("general") && isWarehouse) continue;
200                  if (filter.startsWith("wh:") && !filter.equals("wh:" + row.warehouseName())) continue;
201              }
202              hasRows = true;
203  
204              HBox r = new HBox(4);
205              r.setStyle("-fx-background-color: " + (odd ? "#ffffff" : "#fafaf8") +
206                      "; -fx-padding: 8 10; -fx-border-color: transparent transparent #f0f0f0 transparent; -fx-cursor: hand;");
207              odd = !odd;
208  
209              Label numLbl = new Label(row.supplier().getSupplierNo() > 0 ? String.valueOf(row.supplier().getSupplierNo()) : "—");
210              numLbl.setStyle("-fx-font-size: 12px; -fx-font-weight: bold; -fx-text-fill: #5f5e5a;");
211              numLbl.setMinWidth(widths[0]); numLbl.setPrefWidth(widths[0]);
212  
213              Label nameLbl = new Label(row.supplier().getName());
214              nameLbl.setStyle("-fx-font-size: 13px; -fx-font-weight: bold; -fx-text-fill: #1a1a18;");
215              nameLbl.setMinWidth(widths[1]); nameLbl.setPrefWidth(widths[1]);
216  
217              Label phoneLbl = new Label(row.supplier().getPhone() != null && !row.supplier().getPhone().isEmpty() ?
218                      row.supplier().getPhone() : "—");
219              phoneLbl.setStyle("-fx-font-size: 12px; -fx-text-fill: #5f5e5a;");
220              phoneLbl.setMinWidth(widths[2]); phoneLbl.setPrefWidth(widths[2]);
221  
222              Label sectorLbl = new Label(row.supplier().getSector() != null && !row.supplier().getSector().isEmpty() ?
223                      row.supplier().getSector() : "—");
224              sectorLbl.setStyle("-fx-font-size: 12px; -fx-text-fill: #5f5e5a;");
225              sectorLbl.setMinWidth(widths[3]); sectorLbl.setPrefWidth(widths[3]);
226  
227              Label floorLbl = new Label(String.format("%.0f جنيه", row.supplier().getFloorAmount()));
228              floorLbl.setStyle("-fx-font-size: 12px; -fx-text-fill: #1a1a18;");
229              floorLbl.setMinWidth(widths[4]); floorLbl.setPrefWidth(widths[4]);
230  
231              Label typeLbl;
232              if (isWarehouse) {
233                  typeLbl = new Label("🏭 " + row.warehouseName());
234                  typeLbl.setStyle("-fx-font-size: 11px; -fx-padding: 3 10; -fx-background-radius: 4; " +
235                          "-fx-background-color: #EAF3DE; -fx-text-fill: #3B6D11;");
236              } else {
237                  typeLbl = new Label("🌍 مورد عام");
238                  typeLbl.setStyle("-fx-font-size: 11px; -fx-padding: 3 10; -fx-background-radius: 4; " +
239                          "-fx-background-color: #E6F1FB; -fx-text-fill: #185FA5;");
240              }
241              typeLbl.setMinWidth(widths[5]); typeLbl.setPrefWidth(widths[5]);
242  
243              Button openBtn = new Button("فتح الحساب");
244              openBtn.setStyle("-fx-background-color: #3B6D11; -fx-text-fill: white; -fx-font-size: 11px; " +
245                      "-fx-padding: 4 10; -fx-background-radius: 4; -fx-cursor: hand;");
246              openBtn.setMinWidth(widths[6]); openBtn.setPrefWidth(widths[6]);
247              final SupplierRow finalRow = row;
248              openBtn.setOnAction(e -> {
249                  MainLayout.loadContent(SupplierDetailContent.build(userId, username, role, finalRow.supplier()));
250                  MainLayout.setTitle("حساب: " + finalRow.supplier().getName());
251              });
252  
253              Button editBtn = new Button("تعديل");
254              editBtn.setStyle("-fx-background-color: #EAF3DE; -fx-text-fill: #3B6D11; -fx-font-size: 11px; " +
255                      "-fx-padding: 4 8; -fx-background-radius: 4; -fx-cursor: hand;");
256              editBtn.setMinWidth(widths[7]); editBtn.setPrefWidth(widths[7]);
257              editBtn.setDisable(!role.equals("admin"));
258              editBtn.setOnAction(e -> showEditDialog(userId, username, role, finalRow, rows, table, filter));
259  
260              Button deleteBtn = new Button("حذف");
261              deleteBtn.setStyle("-fx-background-color: #FCEBEB; -fx-text-fill: #A32D2D; -fx-font-size: 11px; " +
262                      "-fx-padding: 4 8; -fx-background-radius: 4; -fx-cursor: hand;");
263              deleteBtn.setMinWidth(widths[8]); deleteBtn.setPrefWidth(widths[8]);
264              deleteBtn.setDisable(!role.equals("admin"));
265              deleteBtn.setOnAction(e -> {
266                  String msg = isWarehouse
267                          ? "هتحذف \"" + finalRow.supplier().getName() + "\" نهائيًا وهيتشال من مخزن \"" + finalRow.warehouseName() + "\"؟"
268                          : "هتحذف \"" + finalRow.supplier().getName() + "\" نهائيًا؟ سجلات الشحنات بتاعته هتفضل موجودة.";
269                  Alert confirm = new Alert(Alert.AlertType.CONFIRMATION, msg);
270                  confirm.showAndWait().ifPresent(res -> {
271                      if (res == ButtonType.OK) {
272                          deleteBtn.setDisable(true);
273                          AsyncHelper.run(
274                                  () -> {
275                                      SupplierDAO.deleteSupplierFully(finalRow.supplier().getId());
276                                      List<SupplierRow> freshRows = new ArrayList<>();
277                                      loadSupplierRows(freshRows, userId, role);
278                                      return freshRows;
279                                  },
280                                  freshRows -> {
281                                      rows.clear();
282                                      rows.addAll(freshRows);
283                                      renderTable(table, rows, filter, userId, username, role);
284                                  },
285                                  error -> {
286                                      deleteBtn.setDisable(false);
287                                      new Alert(Alert.AlertType.ERROR, "حصل خطأ أثناء الحذف").showAndWait();
288                                  }
289                          );
290                      }
291                  });
292              });
293  
294              r.getChildren().addAll(numLbl, nameLbl, phoneLbl, sectorLbl, floorLbl, typeLbl, openBtn, editBtn, deleteBtn);
295              table.getChildren().add(r);
296          }
297  
298          if (!hasRows) {
299              Label empty = new Label("مفيش موردين");
300              empty.setStyle("-fx-text-fill: #888780; -fx-padding: 20; -fx-font-size: 13px;");
301              table.getChildren().add(empty);
302          }
303      }

305      private static void showAddDialog(int userId, String username, String role,
306                                        List<SupplierRow> allRows, VBox tableBody,
307                                        List<Button> allFilterBtns) {
308          TextField nameField = new TextField(); nameField.setPromptText("اسم المورد");
309          TextField phoneField = new TextField(); phoneField.setPromptText("رقم التليفون");
310          TextField sectorField = new TextField(); sectorField.setPromptText("القطاع");
311          TextField floorField = new TextField(); floorField.setPromptText("الأرضية");
312  
313          List<Warehouse> warehouses = WarehouseDAO.getAllWarehouses();
314          ComboBox<String> warehouseCombo = new ComboBox<>();
315          warehouseCombo.getItems().add("مورد عام (بدون مخزن)");
316          for (Warehouse w : warehouses) warehouseCombo.getItems().add(w.getId() + "|" + w.getName());
317          warehouseCombo.setValue("مورد عام (بدون مخزن)");
318          warehouseCombo.setMaxWidth(Double.MAX_VALUE);
319  
320          Button saveBtn = new Button("حفظ"); saveBtn.getStyleClass().add("btn-primary");
321          Label errorLbl = new Label("");
322  
323          VBox layout = new VBox(10,
324                  new Label("الاسم:"), nameField,
325                  new Label("التليفون:"), phoneField,
326                  new Label("القطاع:"), sectorField,
327                  new Label("الأرضية:"), floorField,
328                  new Label("المخزن:"), warehouseCombo,
329                  saveBtn, errorLbl);
330          layout.setPadding(new Insets(20));
331  
332          Stage dialog = DialogHelper.create("إضافة مورد جديد", layout, 360, 380);
333  
334          saveBtn.setOnAction(e -> {
335              String name = nameField.getText().trim();
336              if (name.isEmpty()) { errorLbl.setText("الاسم مطلوب"); return; }
337              double floor = 0;
338              try {
339                  if (!floorField.getText().trim().isEmpty())
340                      floor = Double.parseDouble(floorField.getText().trim());
341              } catch (NumberFormatException ex) { errorLbl.setText("الأرضية لازم رقم"); return; }
342  
343              final String phone = phoneField.getText().trim();
344              final String sector = sectorField.getText().trim();
345              final double finalFloor = floor;
346              final String selected = warehouseCombo.getValue();
347  
348              errorLbl.setStyle("-fx-text-fill: #5f5e5a;");
349              errorLbl.setText("جاري الحفظ...");
350  
351              // ── الحفظ + إعادة تحميل الجدول في الخلفية عشان الـ dialog ميهنجش ──
352              AsyncHelper.run(
353                      () -> {
354                          SupplierDAO.addSupplier(name, phone, sector, finalFloor, userId);
355  
356                          if (selected != null && selected.contains("|")) {
357                              int warehouseId = Integer.parseInt(selected.split("\\|")[0]);
358                              List<Supplier> all = SupplierDAO.getAllSuppliers();
359                              for (Supplier s : all) {
360                                  if (s.getName().equals(name)) {
361                                      WarehouseDAO.assignSupplierToWarehouse(warehouseId, s.getId());
362                                      break;
363                                  }
364                              }
365                          }
366  
367                          List<SupplierRow> freshRows = new ArrayList<>();
368                          loadSupplierRows(freshRows, userId, role);
369                          return freshRows;
370                      },
371                      freshRows -> {
372                          allRows.clear();
373                          allRows.addAll(freshRows);
374                          renderTable(tableBody, allRows, null, userId, username, role);
375                          dialog.close();
376                      },
377                      error -> {
378                          errorLbl.setStyle("-fx-text-fill: #b91c1c;");
379                          errorLbl.setText("حصل خطأ أثناء الحفظ، حاول تاني");
380                      },
381                      saveBtn
382              );
383          });
384  
385          dialog.show();
386      }

388      private static void showEditDialog(int userId, String username, String role, SupplierRow row,
389                                         List<SupplierRow> allRows, VBox tableBody, String filter) {
390          Supplier supplier = row.supplier();
391  
392          TextField nameField = new TextField(supplier.getName());
393          TextField phoneField = new TextField(supplier.getPhone() != null ? supplier.getPhone() : "");
394          TextField sectorField = new TextField(supplier.getSector() != null ? supplier.getSector() : "");
395  
396          ComboBox<String> warehouseCombo = new ComboBox<>();
397          warehouseCombo.getItems().add("مورد عام (بدون مخزن)");
398          warehouseCombo.setMaxWidth(Double.MAX_VALUE);
399          warehouseCombo.setDisable(true);
400          warehouseCombo.setPromptText("جاري تحميل المخازن...");
401  
402          Button saveBtn = new Button("حفظ"); saveBtn.getStyleClass().add("btn-primary");
403          Label errorLbl = new Label("");
404  
405          VBox layout = new VBox(10,
406                  new Label("الاسم:"), nameField,
407                  new Label("التليفون:"), phoneField,
408                  new Label("القطاع:"), sectorField,
409                  new Label("المخزن:"), warehouseCombo,
410                  saveBtn, errorLbl);
411          layout.setPadding(new Insets(20));
412          Stage dialog = DialogHelper.create("تعديل مورد", layout, 360, 400);
413  
414          // تحميل المخازن + المخزن الحالي بتاع المورد في الخلفية
415          AsyncHelper.run(
416                  () -> new Object[]{WarehouseDAO.getAllWarehouses(), WarehouseDAO.getWarehouseBySupplier(supplier.getId())},
417                  result -> {
418                      @SuppressWarnings("unchecked")
419                      List<Warehouse> warehouses = (List<Warehouse>) result[0];
420                      int currentWarehouseId = (int) result[1];
421                      for (Warehouse w : warehouses) warehouseCombo.getItems().add(w.getId() + "|" + w.getName());
422                      warehouseCombo.setValue("مورد عام (بدون مخزن)");
423                      if (currentWarehouseId != -1) {
424                          for (String item : warehouseCombo.getItems()) {
425                              if (item.startsWith(currentWarehouseId + "|")) { warehouseCombo.setValue(item); break; }
426                          }
427                      }
428                      warehouseCombo.setDisable(false);
429                  },
430                  error -> warehouseCombo.setPromptText("حصل خطأ في تحميل المخازن")
431          );
432  
433          saveBtn.setOnAction(e -> {
434              String name = nameField.getText().trim();
435              if (name.isEmpty()) { errorLbl.setText("الاسم مطلوب"); return; }
436              String phone = phoneField.getText().trim();
437              String sector = sectorField.getText().trim();
438              String selected = warehouseCombo.getValue();
439  
440              errorLbl.setStyle("-fx-text-fill: #5f5e5a;");
441              errorLbl.setText("جاري الحفظ...");
442  
443              AsyncHelper.run(
444                      () -> {
445                          SupplierDAO.updateSupplier(supplier.getId(), name, phone, sector);
446  
447                          int currentWarehouseId = WarehouseDAO.getWarehouseBySupplier(supplier.getId());
448                          if (currentWarehouseId != -1) {
449                              WarehouseDAO.removeSupplierFromWarehouse(currentWarehouseId, supplier.getId());
450                          }
451                          if (selected != null && selected.contains("|")) {
452                              int newWarehouseId = Integer.parseInt(selected.split("\\|")[0]);
453                              WarehouseDAO.assignSupplierToWarehouse(newWarehouseId, supplier.getId());
454                          }
455  
456                          List<SupplierRow> freshRows = new ArrayList<>();
457                          loadSupplierRows(freshRows, userId, role);
458                          return freshRows;
459                      },
460                      freshRows -> {
461                          allRows.clear();
462                          allRows.addAll(freshRows);
463                          renderTable(tableBody, allRows, filter, userId, username, role);
464                          dialog.close();
465                      },
466                      error -> {
467                          errorLbl.setStyle("-fx-text-fill: #b91c1c;");
468                          errorLbl.setText("حصل خطأ أثناء الحفظ");
469                      },
470                      saveBtn
471              );
472          });
473  
474          dialog.show();
475      }
