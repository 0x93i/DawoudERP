245      private static void refreshStock(Label lbl, int warehouseId) {
246          try (Connection conn = DatabaseManager_online.getConnection()) {
247              // إجمالي الداخل
248              String entrySql = "SELECT COALESCE(SUM(total_weight), 0) as total FROM warehouse_stock_entries WHERE warehouse_id = ?";
249              PreparedStatement es = conn.prepareStatement(entrySql);
250              es.setInt(1, warehouseId);
251              ResultSet ers = es.executeQuery();
252              double totalIn = ers.next() ? ers.getDouble("total") : 0;
253  
254              // إجمالي الخارج بالأنواع
255              String exitSql = """ 
256                  SELECT  
257                      COALESCE(SUM(weight_green), 0) as green, 
258                      COALESCE(SUM(weight_colored), 0) as colored, 
259                      COALESCE(SUM(weight_white), 0) as white, 
260                      COALESCE(SUM(weight_waste), 0) as waste 
261                  FROM warehouse_stock_exits WHERE warehouse_id = ? 
262              """;
263              PreparedStatement xs = conn.prepareStatement(exitSql);
264              xs.setInt(1, warehouseId);
265              ResultSet xrs = xs.executeQuery();
266              double green = 0, colored = 0, white = 0, waste = 0;
267              if (xrs.next()) {
268                  green = xrs.getDouble("green");
269                  colored = xrs.getDouble("colored");
270                  white = xrs.getDouble("white");
271                  waste = xrs.getDouble("waste");
272              }
273              double totalOut = green + colored + white + waste;
274              double remaining = totalIn - totalOut;
275  
276              lbl.setText(String.format(
277                      "داخل: %.1f | خارج: %.1f | متبقي: %.1f كيلو%nأخضر: %.1f | ألوان: %.1f | أبيض: %.1f | زبالة: %.1f",
278                      totalIn, totalOut, remaining, green, colored, white, waste));
279  
280          } catch (SQLException e) {
281              lbl.setText("خطأ في تحميل البيانات");
282          }
283      }
