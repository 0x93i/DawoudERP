88       private static ShipmentEditData loadShipmentEditData(int shipmentId) throws SQLException {
89           try (Connection conn = DatabaseManager_online.getConnection()) {
90               PreparedStatement stmt = conn.prepareStatement("SELECT * FROM shipments WHERE id = ?");
91               stmt.setInt(1, shipmentId);
92               ResultSet rs = stmt.executeQuery();
93               if (!rs.next()) return null;
94   
95               int factoryId = rs.getInt("factory_id");
96               double factoryGross = rs.getDouble("factory_gross_weight");
97               double factoryDed = rs.getDouble("factory_deduction_kg");
98               double factoryPct = factoryGross > 0 ? (factoryDed / factoryGross * 100) : 0;
99               double salePrice = rs.getDouble("sale_price_per_kg");
100              double costLoading = rs.getDouble("cost_loading");
101              double costWorkers = rs.getDouble("cost_workers");
102              double costFuel = rs.getDouble("cost_fuel");
103              double costTransport = rs.getDouble("cost_transport");
104              double costOther = rs.getDouble("cost_other");
105  
106              List<ShipmentSupplierRow> supplierRows = new ArrayList<>();
107              try (PreparedStatement ps = conn.prepareStatement(
108                      "SELECT supplier_id, gross_weight, deduction_pct, price_per_kg FROM shipment_suppliers " +
109                              "WHERE shipment_id = ? ORDER BY id")) {
110                  ps.setInt(1, shipmentId);
111                  ResultSet srs = ps.executeQuery();
112                  while (srs.next()) {
113                      supplierRows.add(new ShipmentSupplierRow(
114                              srs.getInt("supplier_id"), srs.getDouble("gross_weight"),
115                              srs.getDouble("deduction_pct"), srs.getDouble("price_per_kg")));
116                  }
117              }
118  
119              // شحنات قديمة جدًا من قبل ما كان فيه شحنة متعددة الموردين — بيانات المورد الوحيد
120              // كانت متسجلة في جدول shipments نفسه من غير shipment_suppliers
121              if (supplierRows.isEmpty()) {
122                  int supplierId = rs.getInt("supplier_id");
123                  double gross = rs.getDouble("gross_weight");
124                  double ded = rs.getDouble("deduction_kg");
125                  double pct = gross > 0 ? (ded / gross * 100) : 0;
126                  double price = rs.getDouble("price_per_kg");
127                  if (supplierId > 0) supplierRows.add(new ShipmentSupplierRow(supplierId, gross, pct, price));
128              }
129  
130              return new ShipmentEditData(factoryId, supplierRows, factoryGross, factoryPct, salePrice,
131                      costLoading, costWorkers, costFuel, costTransport, costOther);
132          }
133      }

136      private static List<Supplier> loadNonWarehouseSuppliers() {
137          List<Supplier> list = new ArrayList<>();
138          String sql = "SELECT * FROM suppliers WHERE id NOT IN (SELECT supplier_id FROM warehouse_suppliers) ORDER BY supplier_no NULLS LAST, name";
139          try (Connection conn = DatabaseManager_online.getConnection();
140               Statement stmt = conn.createStatement(); ResultSet rs = stmt.executeQuery(sql)) {
141              while (rs.next()) {
142                  list.add(new Supplier(
143                          rs.getInt("id"), rs.getString("name"), rs.getString("phone"),
144                          rs.getString("sector"), rs.getDouble("floor_amount"), rs.getString("floor_date"),
145                          rs.getInt("supplier_no")));
146              }
147          } catch (SQLException e) { System.err.println(e.getMessage()); }
148          return list;
149      }

978      private static List<ShipmentRowData> loadShipmentRows() throws SQLException {
979          List<ShipmentRowData> rows = new ArrayList<>();
980          String sql = """ 
981              SELECT sh.id, sh.shipment_date, 
982                     COALESCE(ss_sup.name, s.name) AS supplier, 
983                     f.name AS factory, 
984                     COALESCE(ss.net_weight,   sh.net_weight)   AS row_net, 
985                     COALESCE(ss.price_per_kg, sh.price_per_kg) AS row_price, 
986                     COALESCE(ss.total_amount, sh.total_amount) AS row_total, 
987                     sh.sale_total, sh.purchase_total, 
988                     sh.cost_loading + sh.cost_workers + sh.cost_fuel 
989                       + sh.cost_transport + sh.cost_other AS total_costs 
990              FROM shipments sh 
991              LEFT JOIN shipment_suppliers ss ON ss.shipment_id = sh.id 
992              LEFT JOIN suppliers ss_sup ON ss.supplier_id = ss_sup.id 
993              LEFT JOIN suppliers s ON sh.supplier_id = s.id 
994              LEFT JOIN factories f ON sh.factory_id = f.id 
995              ORDER BY sh.shipment_date DESC, sh.id DESC, ss.id ASC 
996          """;
997  
998          try (Connection conn = DatabaseManager_online.getConnection();
999               Statement stmt = conn.createStatement();
1000              ResultSet rs = stmt.executeQuery(sql)) {
1001             while (rs.next()) {
1002                 rows.add(new ShipmentRowData(
1003                         rs.getString("shipment_date"),
1004                         rs.getString("supplier"),
1005                         rs.getString("factory"),
1006                         rs.getDouble("row_net"),
1007                         rs.getDouble("row_price"),
1008                         rs.getDouble("row_total"),
1009                         rs.getDouble("sale_total"),
1010                         rs.getDouble("purchase_total"),
1011                         rs.getDouble("total_costs"),
1012                         rs.getInt("id")));
1013             }
1014         }
1015         return rows;
