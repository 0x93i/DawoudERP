16   public class LoginScreen {
17   
18       public static void show(Stage stage) {
19           Label titleLabel = new Label("Dawoud ERP — تسجيل الدخول");
20   
21           TextField usernameField = new TextField();
22           usernameField.setPromptText("اسم المستخدم");
23   
24           PasswordField passwordField = new PasswordField();
25           passwordField.setPromptText("كلمة المرور");
26   
27           Button loginButton = new Button("دخول");
28           Label errorLabel = new Label("");
29   
30           loginButton.setOnAction(e -> {
31               String username = usernameField.getText().trim();
32               String password = passwordField.getText().trim();
33   
34               if (username.isEmpty() || password.isEmpty()) {
35                   errorLabel.setText("من فضلك ادخل اسم المستخدم وكلمة المرور");
36                   return;
37               }
38   
39               String[] result = authenticate(username, password);
40               if (result != null) {
41                   int userId = Integer.parseInt(result[0]);
42                   String role = result[1];
43                   MainLayout.show(stage, userId, username, role);
44               } else {
45                   errorLabel.setText("اسم المستخدم أو كلمة المرور غلط");
46               }
47           });
48   
49           VBox layout = new VBox(12,
50                   titleLabel,
51                   new Label("اسم المستخدم:"), usernameField,
52                   new Label("كلمة المرور:"), passwordField,
53                   loginButton,
54                   errorLabel
55           );
56           layout.setAlignment(Pos.CENTER);
57           layout.setPadding(new Insets(40));
58           layout.setMaxWidth(300);
59   
60           VBox wrapper = new VBox(layout);
61           wrapper.setAlignment(Pos.CENTER);
62   
63           Scene scene = new Scene(wrapper, 800, 600);
64           stage.setScene(scene);
65           scene.getRoot().setNodeOrientation(javafx.geometry.NodeOrientation.RIGHT_TO_LEFT);
66           stage.setTitle("برنامج عم داود");
67           stage.show();
68       }
69   
70       private static String[] authenticate(String username, String password) {
71           String sql = "SELECT id, role FROM users WHERE username = ? AND password_hash = ?";
72           try (Connection conn = DatabaseManager_online.getConnection();
73                PreparedStatement stmt = conn.prepareStatement(sql)) {
74               stmt.setString(1, username);
75               stmt.setString(2, password);
76               ResultSet rs = stmt.executeQuery();
77               if (rs.next()) {
78                   return new String[]{
79                           String.valueOf(rs.getInt("id")),
80                           rs.getString("role")
81                   };
82               }
83           } catch (SQLException e) {
84               System.err.println("Auth error: " + e.getMessage());
85           }
86           return null;
87       }
88   }
