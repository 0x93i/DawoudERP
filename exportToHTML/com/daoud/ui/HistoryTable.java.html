13       public static class Row {
14           public final String[] values;
15           public final String type;
16           public Row(String type, String... values) {
17               this.type = type;
18               this.values = values;
19           }
20       }

22       public static VBox build(String[] headers, double[] widths, List<Row> allRows, String[] filterTypes) {
23   
24           // ── الفلاتر ──
25           HBox filtersBox = new HBox(8);
26           filtersBox.setAlignment(Pos.CENTER_RIGHT);
27   
28           VBox tableBox = new VBox(0);
29           tableBox.setStyle("-fx-border-color: #e0e0e0; -fx-border-radius: 0 0 6 6;");
30   
31           // زر "الكل"
32           Button allBtn = filterBtn("الكل", true);
33           filtersBox.getChildren().add(allBtn);
34   
35           // أزرار الفلاتر
36           for (String type : filterTypes) {
37               Button btn = filterBtn(type, false);
38               filtersBox.getChildren().add(btn);
39           }
40   
41           // رسم الجدول
42           Runnable[] renderAll = {null};
43           renderAll[0] = () -> renderTable(tableBox, headers, widths, allRows, null);
44           renderAll[0].run();
45   
46           // أحداث الفلاتر
47           allBtn.setOnAction(e -> {
48               setActive(allBtn, filtersBox);
49               renderTable(tableBox, headers, widths, allRows, null);
50           });
51   
52           for (int i = 0; i < filterTypes.length; i++) {
53               final String type = filterTypes[i];
54               Button btn = (Button) filtersBox.getChildren().get(i + 1);
55               btn.setOnAction(e -> {
56                   setActive(btn, filtersBox);
57                   renderTable(tableBox, headers, widths, allRows, r -> r.type.equals(type));
58               });
59           }
60   
61           // ── Header ──
62           HBox header = buildHeader(headers, widths);
63   
64           VBox card = new VBox(0);
65           card.getStyleClass().add("card");
66   
67           HBox topBar = new HBox(filtersBox);
68           topBar.setAlignment(Pos.CENTER_RIGHT);
69           topBar.setStyle("-fx-padding: 0 0 10 0;");
70   
71           card.getChildren().addAll(topBar, header, tableBox);
72           return card;
73       }
74   
75       private static void renderTable(VBox table, String[] headers, double[] widths,
76                                       List<Row> rows, Predicate<Row> filter) {
77           table.getChildren().clear();
78           List<Row> filtered = filter == null ? rows : rows.stream().filter(filter).collect(Collectors.toList());
79   
80           if (filtered.isEmpty()) {
81               Label empty = new Label("مفيش بيانات");
82               empty.setStyle("-fx-text-fill: #888780; -fx-padding: 16; -fx-font-size: 13px;");
83               table.getChildren().add(empty);
84               return;
85           }
86   
87           boolean odd = true;
88           for (Row row : filtered) {
89               HBox r = new HBox();
90               String bg = odd ? "#ffffff" : "#fafaf8";
91               r.setStyle("-fx-background-color: " + bg + "; -fx-padding: 8 10; -fx-border-color: transparent transparent #f0f0f0 transparent;");
92               odd = !odd;
93   
94               for (int i = 0; i < row.values.length && i < widths.length; i++) {
95                   Label lbl = new Label(row.values[i]);
96                   lbl.setStyle("-fx-font-size: 12px; -fx-text-fill: #1a1a18;");
97                   lbl.setMinWidth(widths[i]); lbl.setPrefWidth(widths[i]);
98                   r.getChildren().add(lbl);
99               }
100              table.getChildren().add(r);
101          }
102      }
103  
104      private static HBox buildHeader(String[] headers, double[] widths) {
105          HBox header = new HBox();
106          header.setStyle("-fx-background-color: #f5f5f3; -fx-padding: 8 10; -fx-background-radius: 6 6 0 0;");
107          for (int i = 0; i < headers.length; i++) {
108              Label lbl = new Label(headers[i]);
109              lbl.setStyle("-fx-font-size: 11px; -fx-font-weight: bold; -fx-text-fill: #888780;");
110              lbl.setMinWidth(widths[i]); lbl.setPrefWidth(widths[i]);
111              header.getChildren().add(lbl);
112          }
113          return header;
114      }
115  
116      private static Button filterBtn(String text, boolean active) {
117          Button btn = new Button(text);
118          btn.getStyleClass().add(active ? "btn-primary" : "btn-default");
119          btn.setStyle(btn.getStyle() + "-fx-font-size: 11px; -fx-padding: 4 10;");
120          return btn;
121      }
122  
123      private static void setActive(Button active, HBox box) {
124          for (javafx.scene.Node n : box.getChildren()) {
125              if (n instanceof Button b) {
126                  b.getStyleClass().removeAll("btn-primary", "btn-default");
127                  b.getStyleClass().add(b == active ? "btn-primary" : "btn-default");
128              }
129          }
130      }
