1    package com.daoud.ui;
2    
3    import com.daoud.dao.CustodyDAO;
4    import com.daoud.dao.WarehouseDAO;
5    import com.daoud.model.Supplier;
6    import com.daoud.model.Warehouse;
7    import javafx.collections.FXCollections;
8    import javafx.geometry.Insets;
9    import javafx.scene.Scene;
10   import javafx.scene.control.*;
11   import javafx.scene.layout.*;
12   import javafx.stage.Stage;
13   
14   import java.util.List;
15   
16   public class CustodyScreen {
17   
18       public static void show(Stage stage, int userId, String username, String role) {
19   
20           // ── التاريخ (بيتعرفوا أول عشان الـ listeners تشوفهم) ──
21           TextArea custodyHistory = new TextArea();
22           custodyHistory.setEditable(false);
23           custodyHistory.setPrefHeight(120);
24   
25           TextArea disburseHistory = new TextArea();
26           disburseHistory.setEditable(false);
27           disburseHistory.setPrefHeight(120);
28   
29           // ── اختيار المخزن ──
30           List<Warehouse> warehouses = WarehouseDAO.getAllWarehouses();
31           ComboBox<Warehouse> warehouseCombo = new ComboBox<>(
32                   FXCollections.observableArrayList(warehouses));
33           warehouseCombo.setPromptText("اختار المخزن");
34   
35           Label remainingLabel = new Label("المتبقي من العهد: —");
36   
37           // ── إضافة عهدة (أدمن بس) ──
38           Label custodyTitle = new Label("── إضافة عهدة جديدة (أدمن فقط) ──");
39           TextField custodyAmount = new TextField();
40           custodyAmount.setPromptText("المبلغ");
41           TextField custodyNotes = new TextField();
42           custodyNotes.setPromptText("ملاحظة (اختياري)");
43           Button addCustodyBtn = new Button("إضافة عهدة");
44           Label custodyMsg = new Label("");
45   
46           custodyTitle.setVisible(role.equals("admin"));
47           custodyTitle.setManaged(role.equals("admin"));
48           custodyAmount.setVisible(role.equals("admin"));
49           custodyAmount.setManaged(role.equals("admin"));
50           custodyNotes.setVisible(role.equals("admin"));
51           custodyNotes.setManaged(role.equals("admin"));
52           addCustodyBtn.setVisible(role.equals("admin"));
53           addCustodyBtn.setManaged(role.equals("admin"));
54           custodyMsg.setVisible(role.equals("admin"));
55           custodyMsg.setManaged(role.equals("admin"));
56   
57           // ── توزيع ──
58           Label disburseTitle = new Label("── تسجيل توزيع من العهدة ──");
59   
60           ComboBox<String> recipientTypeCombo = new ComboBox<>(
61                   FXCollections.observableArrayList("مورد", "أخرى"));
62           recipientTypeCombo.setPromptText("نوع المستلم");
63   
64           ComboBox<Supplier> supplierCombo = new ComboBox<>();
65           supplierCombo.setPromptText("اختار المورد");
66           supplierCombo.setVisible(false);
67           supplierCombo.setManaged(false);
68   
69           TextField otherNotes = new TextField();
70           otherNotes.setPromptText("وصف (مثلاً: بنزين)");
71           otherNotes.setVisible(false);
72           otherNotes.setManaged(false);
73   
74           TextField disburseAmount = new TextField();
75           disburseAmount.setPromptText("المبلغ");
76   
77           Button disburseBtn = new Button("تسجيل التوزيع");
78           Label disburseMsg = new Label("");
79   
80           // ── الأحداث ──
81           recipientTypeCombo.setOnAction(e -> {
82               String type = recipientTypeCombo.getValue();
83               supplierCombo.setVisible("مورد".equals(type));
84               supplierCombo.setManaged("مورد".equals(type));
85               otherNotes.setVisible("أخرى".equals(type));
86               otherNotes.setManaged("أخرى".equals(type));
87           });
88   
89           warehouseCombo.setOnAction(e -> {
90               Warehouse w = warehouseCombo.getValue();
91               if (w == null) return;
92               remainingLabel.setText(String.format("المتبقي من العهد: %.1f جنيه",
93                       CustodyDAO.getRemainingCustody(w.getId())));
94               supplierCombo.setItems(FXCollections.observableArrayList(
95                       WarehouseDAO.getSuppliersByWarehouse(w.getId())));
96               refreshHistories(custodyHistory, disburseHistory, w.getId());
97           });
98   
99           addCustodyBtn.setOnAction(e -> {
100              Warehouse w = warehouseCombo.getValue();
101              if (w == null) { custodyMsg.setText("اختار مخزن"); return; }
102              try {
103                  double amount = Double.parseDouble(custodyAmount.getText().trim());
104                  CustodyDAO.addCustody(w.getId(), userId, amount, custodyNotes.getText().trim());
105                  remainingLabel.setText(String.format("المتبقي من العهد: %.1f جنيه",
106                          CustodyDAO.getRemainingCustody(w.getId())));
107                  custodyAmount.clear();
108                  custodyNotes.clear();
109                  custodyMsg.setText("تم إضافة العهدة");
110                  refreshHistories(custodyHistory, disburseHistory, w.getId());
111              } catch (NumberFormatException ex) {
112                  custodyMsg.setText("ادخل رقم صحيح");
113              }
114          });
115  
116          disburseBtn.setOnAction(e -> {
117              Warehouse w = warehouseCombo.getValue();
118              if (w == null) { disburseMsg.setText("اختار مخزن"); return; }
119              if (recipientTypeCombo.getValue() == null) { disburseMsg.setText("اختار نوع المستلم"); return; }
120              try {
121                  double amount = Double.parseDouble(disburseAmount.getText().trim());
122                  double remaining = CustodyDAO.getRemainingCustody(w.getId());
123                  if (amount > remaining) { disburseMsg.setText("المبلغ أكبر من المتبقي"); return; }
124  
125                  String type = recipientTypeCombo.getValue().equals("مورد") ? "supplier" : "other";
126                  int recipientId = 0;
127                  String notes = "";
128  
129                  if (type.equals("supplier")) {
130                      Supplier s = supplierCombo.getValue();
131                      if (s == null) { disburseMsg.setText("اختار المورد"); return; }
132                      recipientId = s.getId();
133                      notes = s.getName();
134                  } else {
135                      notes = otherNotes.getText().trim();
136                  }
137  
138                  CustodyDAO.addDisbursement(w.getId(), userId, type, recipientId, amount, notes);
139                  remainingLabel.setText(String.format("المتبقي من العهد: %.1f جنيه",
140                          CustodyDAO.getRemainingCustody(w.getId())));
141                  disburseAmount.clear();
142                  disburseMsg.setText("تم التسجيل");
143                  refreshHistories(custodyHistory, disburseHistory, w.getId());
144              } catch (NumberFormatException ex) {
145                  disburseMsg.setText("ادخل رقم صحيح");
146              }
147          });
148  
149          Button backBtn = new Button("رجوع");
150          backBtn.setOnAction(e -> MainScreen.show(stage, userId, username, role));
151  
152          VBox layout = new VBox(10,
153                  new Label("العهد"),
154                  new Label("المخزن:"), warehouseCombo,
155                  remainingLabel,
156                  new Separator(),
157                  custodyTitle,
158                  new Label("المبلغ:"), custodyAmount,
159                  new Label("ملاحظة:"), custodyNotes,
160                  addCustodyBtn, custodyMsg,
161                  new Separator(),
162                  disburseTitle,
163                  new Label("نوع المستلم:"), recipientTypeCombo,
164                  supplierCombo, otherNotes,
165                  new Label("المبلغ:"), disburseAmount,
166                  disburseBtn, disburseMsg,
167                  new Separator(),
168                  new Label("── سجل العهد ──"), custodyHistory,
169                  new Label("── سجل التوزيع ──"), disburseHistory,
170                  backBtn
171          );
172          layout.setPadding(new Insets(20));
173  
174          ScrollPane scroll = new ScrollPane(layout);
175          scroll.setFitToWidth(true);
176          stage.setScene(new Scene(scroll, 800, 600));
177      }
178  
179      private static void refreshHistories(TextArea custodyArea, TextArea disburseArea, int warehouseId) {
180          custodyArea.setText(String.join("\n", CustodyDAO.getCustodyHistory(warehouseId)));
181          disburseArea.setText(String.join("\n", CustodyDAO.getDisbursementHistory(warehouseId)));
182      }
183  }
