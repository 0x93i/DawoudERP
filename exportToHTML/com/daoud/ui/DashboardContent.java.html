44       private static Node buildContent(DashboardData data) {
45           Label totalInLabel = new Label(String.format("%.0f", data.stats().totalIn()));
46           Label totalOutLabel = new Label(String.format("%.0f", data.stats().totalOut()));
47           Label balanceLabel = new Label(String.format("%.0f", data.stats().balance()));
48           Label factoryDebtLabel = new Label(String.format("%.0f", data.stats().factoryDebt()));
49           factoryDebtLabel.setStyle("-fx-font-size: 22px; -fx-font-weight: bold; -fx-text-fill: " +
50                   (data.stats().factoryDebt() > 0 ? "#3B6D11" : "#A32D2D") + ";");
51   
52           HBox statsRow = new HBox(12,
53                   statCard("إجمالي الداخل (هذا الأسبوع)", totalInLabel, "كيلو"),
54                   statCard("إجمالي الخارج للمصانع", totalOutLabel, "كيلو"),
55                   statCard("رصيد المخازن الحالي", balanceLabel, "كيلو"),
56                   statCard("مستحقات من المصانع", factoryDebtLabel, "جنيه")
57           );
58           statsRow.getChildren().forEach(n -> HBox.setHgrow(n, Priority.ALWAYS));
59   
60           // ── آخر المعاملات ──
61           VBox transCard = new VBox(8); transCard.getStyleClass().add("card");
62           Label transTitle = new Label("آخر المعاملات"); transTitle.getStyleClass().add("card-title");
63           VBox transList = new VBox(4);
64           for (TransactionRow t : data.transactions()) transList.getChildren().add(transactionRow(t));
65           transCard.getChildren().addAll(transTitle, transList);
66   
67           // ── أرصدة الموردين ──
68           VBox suppCard = new VBox(8); suppCard.getStyleClass().add("card");
69           Label suppTitle = new Label("أرصدة الموردين"); suppTitle.getStyleClass().add("card-title");
70           VBox suppList = new VBox(4);
71           for (SupplierBalanceRow b : data.balances()) suppList.getChildren().add(balanceRow(b));
72           suppCard.getChildren().addAll(suppTitle, suppList);
73   
74           HBox bottomRow = new HBox(12, suppCard, transCard);
75           HBox.setHgrow(transCard, Priority.ALWAYS);
76           HBox.setHgrow(suppCard, Priority.ALWAYS);
77   
78           VBox content = new VBox(14, statsRow, bottomRow);
79           content.setPadding(new Insets(4));
80           return content;
81       }

154      private static Stats loadStats() throws SQLException {
155          double totalIn = 0, totalOut = 0, in = 0, out = 0, ships = 0, pays = 0;
156  
157          try (Connection conn = DatabaseManager_online.getConnection()) {
158  
159              try (PreparedStatement ps = conn.prepareStatement(
160                      "SELECT COALESCE(SUM(total_weight), 0) " +
161                              "FROM warehouse_stock_entries " +
162                              "WHERE entry_date::date >= CURRENT_DATE - INTERVAL '7 days'");
163                   ResultSet rs = ps.executeQuery()) {
164                  if (rs.next()) totalIn = rs.getDouble(1);
165              }
166  
167              try (PreparedStatement ps = conn.prepareStatement(
168                      "SELECT COALESCE((SUM(weight_green)+SUM(weight_colored)+SUM(weight_white)+SUM(weight_waste)), 0) FROM warehouse_stock_exits");
169                   ResultSet rs = ps.executeQuery()) {
170                  if (rs.next()) totalOut = rs.getDouble(1);
171              }
172  
173              try (PreparedStatement ps = conn.prepareStatement(
174                      "SELECT COALESCE(SUM(total_weight), 0) FROM warehouse_stock_entries");
175                   ResultSet rs = ps.executeQuery()) {
176                  if (rs.next()) in = rs.getDouble(1);
177              }
178              try (PreparedStatement ps = conn.prepareStatement(
179                      "SELECT COALESCE((SUM(weight_green)+SUM(weight_colored)+SUM(weight_white)+SUM(weight_waste)), 0) FROM warehouse_stock_exits");
180                   ResultSet rs = ps.executeQuery()) {
181                  if (rs.next()) out = rs.getDouble(1);
182              }
183  
184              try (PreparedStatement ps = conn.prepareStatement(
185                      "SELECT COALESCE(SUM(total_amount), 0) FROM factory_shipments");
186                   ResultSet rs = ps.executeQuery()) {
187                  if (rs.next()) ships = rs.getDouble(1);
188              }
189              try (PreparedStatement ps = conn.prepareStatement(
190                      "SELECT COALESCE(SUM(amount), 0) FROM factory_payments");
191                   ResultSet rs = ps.executeQuery()) {
192                  if (rs.next()) pays = rs.getDouble(1);
193              }
194          }
195  
196          return new Stats(totalIn, totalOut, in - out, ships - pays);
197      }

199      private static List<TransactionRow> loadRecentTransactions() throws SQLException {
200          List<TransactionRow> list = new ArrayList<>();
201          String sql = """ 
202              SELECT 'بضاعة' as type, transaction_date as date, 
203                     s.name as name, 
204                     ((gross_weight - deduction_kg) * price_per_kg) as amount 
205              FROM supplier_transactions st 
206              JOIN suppliers s ON st.supplier_id = s.id 
207              UNION ALL 
208              SELECT 'سحب مورد' as type, withdrawal_date as date, 
209                     s.name as name, amount 
210              FROM supplier_withdrawals sw 
211              JOIN suppliers s ON sw.supplier_id = s.id 
212              UNION ALL 
213              SELECT 'شحنة مصنع' as type, shipment_date as date, 
214                     f.name as name, total_amount as amount 
215              FROM factory_shipments fs 
216              JOIN factories f ON fs.factory_id = f.id 
217              ORDER BY date DESC LIMIT 8 
218          """;
219          try (Connection conn = DatabaseManager_online.getConnection();
220               PreparedStatement stmt = conn.prepareStatement(sql);
221               ResultSet rs = stmt.executeQuery()) {
222              while (rs.next()) {
223                  list.add(new TransactionRow(rs.getString("type"), rs.getString("date"),
224                          rs.getString("name"), rs.getDouble("amount")));
225              }
226          }
227          return list;
228      }

230      private static List<SupplierBalanceRow> loadSupplierBalances() throws SQLException {
231          List<SupplierBalanceRow> list = new ArrayList<>();
232          // استعلام واحد بيحسب رصيد كل مورد (أرضية - مشتريات + سحوبات) بدل ما كان بيعمل
233          // 3 استعلامات منفصلة لكل مورد لوحده (كان ده أبطأ حاجة في اللوحة كلها)
234          String sql = """ 
235              SELECT s.name, 
236                     s.floor_amount 
237                     - COALESCE((SELECT SUM((st.gross_weight - st.deduction_kg) * st.price_per_kg) 
238                                 FROM supplier_transactions st WHERE st.supplier_id = s.id), 0) 
239                     + COALESCE((SELECT SUM(sw.amount) 
240                                 FROM supplier_withdrawals sw WHERE sw.supplier_id = s.id), 0) AS balance 
241              FROM suppliers s 
242              ORDER BY s.name 
243              LIMIT 8 
244          """;
245          try (Connection conn = DatabaseManager_online.getConnection();
246               Statement stmt = conn.createStatement();
247               ResultSet rs = stmt.executeQuery(sql)) {
248              while (rs.next()) {
249                  list.add(new SupplierBalanceRow(rs.getString("name"), rs.getDouble("balance")));
250              }
251          }
252          return list;
253      }
