348      private static List<HistoryRow> loadRows(int supplierId) {
349          List<HistoryRow> rows = new ArrayList<>();
350          String sql1 = "SELECT id, transaction_date, gross_weight, deduction_kg, price_per_kg " +
351                  "FROM supplier_transactions WHERE supplier_id = ? ORDER BY transaction_date DESC LIMIT 30";
352          try (Connection conn = DatabaseManager_online.getConnection();
353               PreparedStatement stmt = conn.prepareStatement(sql1)) {
354              stmt.setInt(1, supplierId);
355              ResultSet rs = stmt.executeQuery();
356              while (rs.next()) {
357                  double gross = rs.getDouble("gross_weight");
358                  double ded = rs.getDouble("deduction_kg");
359                  double price = rs.getDouble("price_per_kg");
360                  double net = gross - ded;
361                  double pct = gross > 0 ? (ded / gross * 100) : 0;
362                  rows.add(new HistoryRow("بضاعة", rs.getInt("id"), rs.getString("transaction_date"),
363                          String.format("%.1f كيلو", net), String.format("%.0f جنيه", net * price),
364                          String.format("%.1f%%", pct), "—", ""));
365              }
366          } catch (SQLException e) { System.err.println(e.getMessage()); }
367  
368          String sql2 = "SELECT id, withdrawal_date, amount, payment_method, notes " +
369                  "FROM supplier_withdrawals WHERE supplier_id = ? ORDER BY withdrawal_date DESC LIMIT 30";
370          try (Connection conn = DatabaseManager_online.getConnection();
371               PreparedStatement stmt = conn.prepareStatement(sql2)) {
372              stmt.setInt(1, supplierId);
373              ResultSet rs = stmt.executeQuery();
374              while (rs.next()) {
375                  rows.add(new HistoryRow("سحب", rs.getInt("id"), rs.getString("withdrawal_date"),
376                          "—", String.format("%.0f جنيه", rs.getDouble("amount")),
377                          "—", rs.getString("payment_method") != null ? rs.getString("payment_method") : "كاش",
378                          rs.getString("notes") != null ? rs.getString("notes") : ""));
379              }
380          } catch (SQLException e) { System.err.println(e.getMessage()); }
381  
382          // سحوبات "من رصيد الأرضية" — بتظهر هنا للعرض بس، ومش داخلة في حساب الرصيد فوق
383          for (com.daoud.dao.FloorDAO.FloorTransaction ft : com.daoud.dao.FloorDAO.getHistory(supplierId)) {
384              if (!ft.direction().equals("out")) continue;
385              rows.add(new HistoryRow("أرضية", ft.id(), ft.date(),
386                      "—", String.format("%.0f جنيه", ft.amount()),
387                      "—", "من الأرضية", ft.notes()));
388          }
389  
390          rows.sort((a, b) -> b.date().compareTo(a.date()));
391          return rows;
392      }
