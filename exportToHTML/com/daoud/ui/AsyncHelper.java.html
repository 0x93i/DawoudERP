6    import java.util.function.Consumer;

8    /**
9     * مساعد بسيط لتشغيل استعلامات قاعدة البيانات (أو أي شغل بطيء) في thread منفصل
10    * عن الواجهة، عشان البرنامج ميهنجش وهو بيجيب أو بيسجل بيانات من Supabase.
11    *
12    * الاستخدام:
13    * <pre>
14    *   AsyncHelper.run(
15    *       () -> SupplierDAO.getAllSuppliers(),   // الشغل البطيء - يتنفذ في الخلفية
16    *       result -> renderTable(result),         // لما ينجح - يتنفذ على واجهة المستخدم
17    *       error -> DialogHelper... ,              // (اختياري) لو فشل
18    *       saveBtn                                 // (اختياري) عناصر تتعطل أثناء التنفيذ
19    *   );
20    * </pre>
21    */
22   public class AsyncHelper {
23   
24       /** شغل بيرجع نتيجة (زي DAO.getAll()) وممكن يرمي Exception عادي (مش لازم checked خاص). */
25       public interface Work<T> {
26           T call() throws Exception;
27       }
28   
29       /** شغل من غير نتيجة (زي DAO.insert()) وممكن يرمي Exception عادي. */
30       public interface VoidWork {
31           void call() throws Exception;
32       }
33   
34       // ── نسخة بترجع نتيجة ──
35   
36       public static <T> void run(Work<T> backgroundWork, Consumer<T> onSuccess, Node... disableWhileRunning) {
37           run(backgroundWork, onSuccess, null, disableWhileRunning);
38       }
39   
40       public static <T> void run(Work<T> backgroundWork, Consumer<T> onSuccess,
41                                  Consumer<Throwable> onError, Node... disableWhileRunning) {
42           setDisabled(disableWhileRunning, true);
43   
44           Task<T> task = new Task<>() {
45               @Override
46               protected T call() throws Exception {
47                   return backgroundWork.call();
48               }
49           };
50   
51           task.setOnSucceeded(e -> {
52               setDisabled(disableWhileRunning, false);
53               if (onSuccess != null) onSuccess.accept(task.getValue());
54           });
55   
56           task.setOnFailed(e -> {
57               setDisabled(disableWhileRunning, false);
58               Throwable ex = task.getException();
59               System.err.println("Async error: " + (ex != null ? ex.getMessage() : "unknown"));
60               if (onError != null) onError.accept(ex);
61           });
62   
63           Thread thread = new Thread(task, "daoud-async");
64           thread.setDaemon(true);
65           thread.start();
66       }
67   
68       // ── نسخة من غير نتيجة (حفظ/حذف/تحديث) ──
69   
70       public static void runVoid(VoidWork backgroundWork, Runnable onSuccess, Node... disableWhileRunning) {
71           runVoid(backgroundWork, onSuccess, null, disableWhileRunning);
72       }
73   
74       public static void runVoid(VoidWork backgroundWork, Runnable onSuccess,
75                                  Consumer<Throwable> onError, Node... disableWhileRunning) {
76           run(() -> { backgroundWork.call(); return null; },
77                   ignored -> { if (onSuccess != null) onSuccess.run(); },
78                   onError, disableWhileRunning);
79       }
80   
81       private static void setDisabled(Node[] nodes, boolean disabled) {
82           if (nodes == null) return;
83           for (Node n : nodes) if (n != null) n.setDisable(disabled);
84       }
