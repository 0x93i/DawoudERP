20       public static void show(Stage stage, int userId, String username, String role) {
21           Label title = new Label("الموردين");
22   
23           ListView<Supplier> listView = new ListView<>();
24           refreshList(listView, userId, role);
25   
26           Button backBtn = new Button("رجوع");
27           backBtn.setOnAction(e -> MainScreen.show(stage, userId, username, role));
28   
29           Button addBtn = new Button("إضافة مورد جديد");
30           Button deleteBtn = new Button("حذف المورد المحدد");
31           Button openBtn = new Button("فتح حساب المورد");
32   
33           // الإضافة والحذف للأدمن بس
34           //addBtn.setDisable(!role.equals("admin"));
35           // الإضافة للكل
36           addBtn.setDisable(false);
37           deleteBtn.setDisable(true);
38           openBtn.setDisable(true);
39   
40           listView.getSelectionModel().selectedItemProperty().addListener((obs, old, selected) -> {
41               boolean hasSelection = selected != null;
42               deleteBtn.setDisable(!hasSelection || !role.equals("admin"));
43               openBtn.setDisable(!hasSelection);
44           });
45   
46           addBtn.setOnAction(e -> showAddDialog(userId, role, listView));
47   
48           deleteBtn.setOnAction(e -> {
49               Supplier selected = listView.getSelectionModel().getSelectedItem();
50               if (selected != null) {
51                   Alert confirm = new Alert(Alert.AlertType.CONFIRMATION,
52                           "هتحذف المورد: " + selected.getName() + "؟");
53                   confirm.showAndWait().ifPresent(response -> {
54                       if (response == ButtonType.OK) {
55                           SupplierDAO.deleteSupplier(selected.getId());
56                           refreshList(listView, userId, role);
57                       }
58                   });
59               }
60           });
61   
62           openBtn.setOnAction(e -> {
63               Supplier selected = listView.getSelectionModel().getSelectedItem();
64               if (selected != null) {
65                   SupplierDetailScreen.show(stage, userId, username, role, selected);
66               }
67           });
68   
69           HBox buttons = new HBox(10, addBtn, openBtn, deleteBtn, backBtn);
70           buttons.setAlignment(Pos.CENTER);
71   
72           VBox layout = new VBox(12, title, listView, buttons);
73           layout.setPadding(new Insets(20));
74   
75           Scene scene = new Scene(layout, 800, 600);
76           stage.setScene(scene);
77       }
78   
79       private static void refreshList(ListView<Supplier> listView, int userId, String role) {
80           List<Supplier> suppliers;
81           if (role.equals("admin")) {
82               suppliers = SupplierDAO.getAllSuppliers();
83           } else {
84               Warehouse w = WarehouseDAO.getWarehouseByManager(userId);
85               if (w != null) {
86                   suppliers = WarehouseDAO.getSuppliersByWarehouse(w.getId());
87               } else {
88                   suppliers = new ArrayList<>();
89               }
90           }
91           listView.setItems(FXCollections.observableArrayList(suppliers));
92       }
93   
94       private static void showAddDialog(int userId, String role, ListView<Supplier> listView) {
95           Stage dialog = new Stage();
96           dialog.setTitle("إضافة مورد جديد");
97   
98           TextField nameField = new TextField();
99           nameField.setPromptText("اسم المورد");
100  
101          TextField sectorField = new TextField();
102          sectorField.setPromptText("القطاع (اختياري)");
103  
104          TextField floorField = new TextField();
105          floorField.setPromptText("الأرضية (اختياري)");
106  
107          Button saveBtn = new Button("حفظ");
108          Label errorLbl = new Label("");
109  
110          saveBtn.setOnAction(e -> {
111              String name = nameField.getText().trim();
112              if (name.isEmpty()) {
113                  errorLbl.setText("اسم المورد مطلوب");
114                  return;
115              }
116              double floor = 0;
117              try {
118                  if (!floorField.getText().trim().isEmpty())
119                      floor = Double.parseDouble(floorField.getText().trim());
120              } catch (NumberFormatException ex) {
121                  errorLbl.setText("الأرضية لازم تكون رقم");
122                  return;
123              }
124  //            SupplierDAO.addSupplier(name, sectorField.getText().trim(), floor, userId);
125              SupplierDAO.addSupplier(name, "", sectorField.getText().trim(), floor, userId);
126              refreshList(listView, userId, role);
127              dialog.close();
128          });
129  
130          VBox layout = new VBox(10,
131                  new Label("اسم المورد:"), nameField,
132                  new Label("القطاع:"), sectorField,
133                  new Label("الأرضية:"), floorField,
134                  saveBtn, errorLbl);
135          layout.setPadding(new Insets(20));
136          layout.setAlignment(Pos.CENTER_LEFT);
137  
138          dialog.setScene(new Scene(layout, 350, 320));
139          dialog.show();
140      }
