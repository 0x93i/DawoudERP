18       public static List<ReportRow> getFactoryReport(int factoryId, LocalDate from, LocalDate to) {
19           List<ReportRow> list = new ArrayList<>();
20           // مقارنة نصية (yyyy-MM-dd) بدل مقارنة تاريخ مباشرة، بنفس الطريقة اللي
21           // حلّت مشكلة كمية الشهر — عشان تفضل شغالة صح مهما كان نوع العمود بالظبط.
22           String sql = "SELECT id, shipment_date, COALESCE(supplier_name,'') as supplier_name, " +
23                   "gross_weight, deduction_pct, deduction_kg, price_per_kg, total_amount, " +
24                   "COALESCE(notes, '') AS notes " +
25                   "FROM factory_shipments WHERE factory_id = ? AND shipment_date::text >= ? AND shipment_date::text <= ? " +
26                   "ORDER BY shipment_date";
27           try (Connection conn = DatabaseManager_online.getConnection();
28                PreparedStatement stmt = conn.prepareStatement(sql)) {
29               stmt.setInt(1, factoryId);
30               stmt.setString(2, from.toString());
31               stmt.setString(3, to.toString());
32               ResultSet rs = stmt.executeQuery();
33               while (rs.next()) {
34                   String supplierName = rs.getString("supplier_name");
35                   list.add(new ReportRow(rs.getInt("id"), rs.getString("shipment_date"), rs.getString("notes"),
36                           supplierName != null && !supplierName.isEmpty() ? supplierName : "—",
37                           rs.getDouble("gross_weight"), rs.getDouble("deduction_pct"),
38                           rs.getDouble("deduction_kg"), rs.getDouble("price_per_kg"),
39                           rs.getDouble("total_amount")));
40               }
41           } catch (SQLException e) {
42               System.err.println("Error: " + e.getMessage());
43           }
44           return list;
45       }

85       public static void addShipment(int factoryId, String supplierName, double grossWeight,
86                                      double deductionPct, double pricePerKg, int recordedBy) {
87           double deductionKg = grossWeight * (deductionPct / 100.0);
88           double netWeight = grossWeight - deductionKg;
89           double total = netWeight * pricePerKg;
90   
91           String sql = """ 
92               INSERT INTO factory_shipments  
93               (factory_id, shipment_date, supplier_name, gross_weight, deduction_pct, deduction_kg, net_weight, price_per_kg, total_amount, recorded_by) 
94               VALUES (?, date('now'), ?, ?, ?, ?, ?, ?, ?, ?) 
95           """;
96           try (Connection conn = DatabaseManager_online.getConnection();
97                PreparedStatement stmt = conn.prepareStatement(sql)) {
98               stmt.setInt(1, factoryId);
99               stmt.setString(2, supplierName);
100              stmt.setDouble(3, grossWeight);
101              stmt.setDouble(4, deductionPct);
102              stmt.setDouble(5, deductionKg);
103              stmt.setDouble(6, netWeight);
104              stmt.setDouble(7, pricePerKg);
105              stmt.setDouble(8, total);
106              stmt.setInt(9, recordedBy);
107              stmt.executeUpdate();
108          } catch (SQLException e) {
109              System.err.println("Error: " + e.getMessage());
110          }
111      }

132      public static double getCurrentMonthNetWeight(int factoryId) {
133          java.time.LocalDate now = java.time.LocalDate.now();
134          // بنحوّل shipment_date لنص ونقارنه بأول 7 حروف من تاريخ اليوم (yyyy-MM)
135          // — بالظبط زي الشكل اللي التاريخ ظاهر بيه في جدول المعاملات — عشان
136          // نضمن إن أي شحنة ظاهرة هناك في الشهر ده تتحسب هنا، مهما كان نوع
137          // عمود التاريخ في قاعدة البيانات.
138          String yearMonth = String.format("%04d-%02d", now.getYear(), now.getMonthValue());
139          double total = 0;
140          String sql = "SELECT COALESCE(SUM(net_weight), 0) FROM factory_shipments " +
141                  "WHERE factory_id = ? AND shipment_date::text LIKE ?";
142          try (Connection conn = DatabaseManager_online.getConnection();
143               PreparedStatement stmt = conn.prepareStatement(sql)) {
144              stmt.setInt(1, factoryId);
145              stmt.setString(2, yearMonth + "%");
146              ResultSet rs = stmt.executeQuery();
147              if (rs.next()) total = rs.getDouble(1);
148          } catch (SQLException e) {
149              System.err.println("Error in getCurrentMonthNetWeight: " + e.getMessage());
150              e.printStackTrace();
151          }
152          return total;
153      }

155      public static double getFactoryBalance(int factoryId) {
156          double totalShipments = 0;
157          double totalPayments = 0;
158          try (Connection conn = DatabaseManager_online.getConnection()) {
159              PreparedStatement s1 = conn.prepareStatement(
160                      "SELECT COALESCE(SUM(total_amount), 0) FROM factory_shipments WHERE factory_id = ?");
161              s1.setInt(1, factoryId);
162              ResultSet r1 = s1.executeQuery();
163              if (r1.next()) totalShipments = r1.getDouble(1);
164  
165              PreparedStatement s2 = conn.prepareStatement(
166                      "SELECT COALESCE(SUM(amount), 0) FROM factory_payments WHERE factory_id = ?");
167              s2.setInt(1, factoryId);
168              ResultSet r2 = s2.executeQuery();
169              if (r2.next()) totalPayments = r2.getDouble(1);
170          } catch (SQLException e) {
171              System.err.println("Error: " + e.getMessage());
172          }
173          // موجب = المصنع مدين لعم داود
174          return totalShipments - totalPayments;
175      }
