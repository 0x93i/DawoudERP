28       public static List<FloorRow> getAllFloors() {
29           List<FloorRow> list = new ArrayList<>();
30           String sql = """ 
31               SELECT s.id, s.name, s.floor_amount, s.supplier_no, 
32                      (SELECT ws.warehouse_id FROM warehouse_suppliers ws WHERE ws.supplier_id = s.id LIMIT 1) AS warehouse_id 
33               FROM suppliers s 
34               ORDER BY s.supplier_no NULLS LAST, s.name 
35           """;
36           try (Connection conn = DatabaseManager_online.getConnection();
37                Statement stmt = conn.createStatement();
38                ResultSet rs = stmt.executeQuery(sql)) {
39               while (rs.next()) {
40                   int warehouseId = rs.getObject("warehouse_id") != null ? rs.getInt("warehouse_id") : -1;
41                   list.add(new FloorRow(rs.getInt("id"), rs.getString("name"), rs.getDouble("floor_amount"),
42                           rs.getInt("supplier_no"), warehouseId));
43               }
44           } catch (SQLException e) {
45               System.err.println("FloorDAO error: " + e.getMessage());
46           }
47           return list;
48       }

63       /**
64        * تسجيل زيادة أو نقصان في رصيد الأرضية بتاع مورد.
65        * direction: "in" (زيادة/إيداع) أو "out" (نقصان/سحب).
66        * مسموح إن الرصيد يبقى سالب لو السحب أكبر من المتاح — من غير أي منع.
67        */
68       public static void adjust(int supplierId, String direction, double amount, String notes, int recordedBy) throws SQLException {
69           double delta = direction.equals("in") ? amount : -amount;
70           try (Connection conn = DatabaseManager_online.getConnection()) {
71               conn.setAutoCommit(false);
72               try {
73                   try (PreparedStatement upd = conn.prepareStatement(
74                           "UPDATE suppliers SET floor_amount = floor_amount + ? WHERE id = ?")) {
75                       upd.setDouble(1, delta);
76                       upd.setInt(2, supplierId);
77                       upd.executeUpdate();
78                   }
79                   try (PreparedStatement ins = conn.prepareStatement(
80                           "INSERT INTO supplier_floor_transactions (supplier_id, transaction_date, direction, amount, notes, recorded_by) " +
81                                   "VALUES (?, CURRENT_DATE, ?, ?, ?, ?)")) {
82                       ins.setInt(1, supplierId);
83                       ins.setString(2, direction);
84                       ins.setDouble(3, amount);
85                       ins.setString(4, notes);
86                       ins.setInt(5, recordedBy);
87                       ins.executeUpdate();
88                   }
89                   conn.commit();
90               } catch (SQLException e) {
91                   conn.rollback();
92                   throw e;
93               } finally {
94                   conn.setAutoCommit(true);
95               }
96           }
97       }

99       public static List<FloorTransaction> getHistory(int supplierId) {
100          List<FloorTransaction> list = new ArrayList<>();
101          String sql = """ 
102              SELECT ft.id, ft.transaction_date, ft.direction, ft.amount, COALESCE(ft.notes,'') AS notes, 
103                     COALESCE(u.username, '') AS recorded_by_name 
104              FROM supplier_floor_transactions ft 
105              LEFT JOIN users u ON ft.recorded_by = u.id 
106              WHERE ft.supplier_id = ? 
107              ORDER BY ft.transaction_date DESC, ft.id DESC 
108              LIMIT 100 
109          """;
110          try (Connection conn = DatabaseManager_online.getConnection();
111               PreparedStatement stmt = conn.prepareStatement(sql)) {
112              stmt.setInt(1, supplierId);
113              ResultSet rs = stmt.executeQuery();
114              while (rs.next()) {
115                  list.add(new FloorTransaction(rs.getInt("id"), rs.getString("transaction_date"),
116                          rs.getString("direction"), rs.getDouble("amount"), rs.getString("notes"),
117                          rs.getString("recorded_by_name")));
118              }
119          } catch (SQLException e) {
120              System.err.println("FloorDAO error: " + e.getMessage());
121          }
122          return list;
123      }
