27       public static double getTotalCustody(int warehouseId) {
28           String sql = "SELECT COALESCE(SUM(amount), 0) FROM custodies WHERE warehouse_id = ?";
29           try (Connection conn = DatabaseManager_online.getConnection();
30                PreparedStatement stmt = conn.prepareStatement(sql)) {
31               stmt.setInt(1, warehouseId);
32               ResultSet rs = stmt.executeQuery();
33               if (rs.next()) return rs.getDouble(1);
34           } catch (SQLException e) {
35               System.err.println("Error: " + e.getMessage());
36           }
37           return 0;
38       }

41       public static double getTotalDisbursed(int warehouseId) {
42           String sql = """ 
43               SELECT COALESCE(SUM(cd.amount), 0) 
44               FROM custody_disbursements cd 
45               JOIN custodies c ON cd.custody_id = c.id 
46               WHERE c.warehouse_id = ? 
47           """;
48           try (Connection conn = DatabaseManager_online.getConnection();
49                PreparedStatement stmt = conn.prepareStatement(sql)) {
50               stmt.setInt(1, warehouseId);
51               ResultSet rs = stmt.executeQuery();
52               if (rs.next()) return rs.getDouble(1);
53           } catch (SQLException e) {
54               System.err.println("Error: " + e.getMessage());
55           }
56           return 0;

65       public static void addDisbursement(int warehouseId, int disbursedByUserId,

67                                          double amount, String notes) {
68           // جيب آخر custody للمخزن ده
69           String getCustody = "SELECT id FROM custodies WHERE warehouse_id = ? ORDER BY id DESC LIMIT 1";
70           try (Connection conn = DatabaseManager_online.getConnection();
71                PreparedStatement stmt = conn.prepareStatement(getCustody)) {
72               stmt.setInt(1, warehouseId);
73               ResultSet rs = stmt.executeQuery();
74               if (rs.next()) {
75                   int custodyId = rs.getInt("id");
76                   String sql = """ 
77                       INSERT INTO custody_disbursements  
78                       (custody_id, disbursed_by_user_id, recipient_type, recipient_id, amount, disbursement_date, notes) 
79                       VALUES (?, ?, ?, ?, ?, date('now'), ?) 
80                   """;
81                   PreparedStatement s2 = conn.prepareStatement(sql);
82                   s2.setInt(1, custodyId);
83                   s2.setInt(2, disbursedByUserId);
84                   s2.setString(3, recipientType);
85                   s2.setInt(4, recipientId);
86                   s2.setDouble(5, amount);
87                   s2.setString(6, notes);
88                   s2.executeUpdate();
89               }
90           } catch (SQLException e) {
91               System.err.println("Error: " + e.getMessage());
92           }

96       public static List<String> getCustodyHistory(int warehouseId) {
97           List<String> list = new ArrayList<>();
98           String sql = """ 
99               SELECT c.given_date, c.amount, c.notes 
100              FROM custodies c 
101              WHERE c.warehouse_id = ? 
102              ORDER BY c.given_date DESC LIMIT 30 
103          """;
104          try (Connection conn = DatabaseManager_online.getConnection();
105               PreparedStatement stmt = conn.prepareStatement(sql)) {
106              stmt.setInt(1, warehouseId);
107              ResultSet rs = stmt.executeQuery();
108              while (rs.next()) {
109                  list.add(String.format("%s | عهدة: %.1f جنيه | %s",
110                          rs.getString("given_date"),
111                          rs.getDouble("amount"),
112                          rs.getString("notes") != null ? rs.getString("notes") : ""));
113              }
114          } catch (SQLException e) {
115              System.err.println("Error: " + e.getMessage());
116          }
117          return list;
118      }

120      // تاريخ التوزيع للمخزن
121      public static List<String> getDisbursementHistory(int warehouseId) {
122          List<String> list = new ArrayList<>();
123          String sql = """ 
124              SELECT cd.disbursement_date, cd.recipient_type, cd.recipient_id, cd.amount, cd.notes 
125              FROM custody_disbursements cd 
126              JOIN custodies c ON cd.custody_id = c.id 
127              WHERE c.warehouse_id = ? 
128              ORDER BY cd.disbursement_date DESC LIMIT 30 
129          """;
130          try (Connection conn = DatabaseManager_online.getConnection();
131               PreparedStatement stmt = conn.prepareStatement(sql)) {
132              stmt.setInt(1, warehouseId);
133              ResultSet rs = stmt.executeQuery();
134              while (rs.next()) {
135                  String type = switch (rs.getString("recipient_type")) {
136                      case "supplier" -> "مورد";
137                      case "worker" -> "عامل";
138                      default -> "أخرى";
139                  };
140                  list.add(String.format("%s | %s | %.1f جنيه | %s",
141                          rs.getString("disbursement_date"),
142                          type,
143                          rs.getDouble("amount"),
144                          rs.getString("notes") != null ? rs.getString("notes") : ""));
145              }
146          } catch (SQLException e) {
147              System.err.println("Error: " + e.getMessage());
148          }
149          return list;
