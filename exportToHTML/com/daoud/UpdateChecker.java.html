1    package com.daoud;
2    
3    import javafx.application.Platform;
4    import javafx.scene.control.Alert;
5    import javafx.scene.control.ButtonType;
6    
7    import java.io.*;
8    import java.net.*;
9    
10   public class UpdateChecker {
11   
12       private static final String CURRENT_VERSION = "3.1.3";
13       private static final String GITHUB_API = "https://api.github.com/repos/0x93i/DawoudERP/releases/latest";
14   
15       public static void check() {
16           new Thread(() -> {
17               try {
18                   URL url = new URL(GITHUB_API);
19                   HttpURLConnection conn = (HttpURLConnection) url.openConnection();
20                   conn.setRequestProperty("Accept", "application/vnd.github.v3+json");
21                   conn.setConnectTimeout(5000);
22                   conn.setReadTimeout(5000);
23   
24                   BufferedReader reader = new BufferedReader(new InputStreamReader(conn.getInputStream()));
25                   StringBuilder sb = new StringBuilder();
26                   String line;
27                   while ((line = reader.readLine()) != null) sb.append(line);
28                   String json = sb.toString();
29   
30                   // استخرج الـ tag_name
31                   int tagStart = json.indexOf("\"tag_name\":\"") + 12;
32                   int tagEnd = json.indexOf("\"", tagStart);
33                   String latestVersion = json.substring(tagStart, tagEnd).replace("v", "");
34   
35                   // استخرج الـ download URL
36                   int urlStart = json.indexOf("\"browser_download_url\":\"") + 24;
37                   int urlEnd = json.indexOf("\"", urlStart);
38                   String downloadUrl = urlStart > 24 ? json.substring(urlStart, urlEnd) : "";
39   
40                   if (!latestVersion.equals(CURRENT_VERSION)) {
41                       final String finalVersion = latestVersion;
42                       final String finalUrl = downloadUrl;
43                       Platform.runLater(() -> showUpdateDialog(finalVersion, finalUrl));
44                   }
45   
46               } catch (Exception e) {
47                   // مفيش انترنت أو GitHub — متعملش حاجة
48                   System.out.println("Update check skipped: " + e.getMessage());
49               }
50           }).start();
51       }
52   
53       private static void showUpdateDialog(String newVersion, String downloadUrl) {
54           Alert alert = new Alert(Alert.AlertType.CONFIRMATION);
55           alert.setTitle("تحديث جديد متاح");
56           alert.setHeaderText("🆕 إصدار جديد: v" + newVersion);
57           alert.setContentText(
58                   "الإصدار الحالي: v" + CURRENT_VERSION + "\n" +
59                           "الإصدار الجديد: v" + newVersion + "\n\n" +
60                           "هل تريد فتح صفحة التحديث؟"
61           );
62   
63           alert.showAndWait().ifPresent(result -> {
64               if (result == ButtonType.OK && !downloadUrl.isEmpty()) {
65                   try {
66                       java.awt.Desktop.getDesktop().browse(new URI(downloadUrl));
67                   } catch (Exception e) {
68                       System.err.println("Cannot open browser: " + e.getMessage());
69                   }
70               }
71           });
72       }
73   }
